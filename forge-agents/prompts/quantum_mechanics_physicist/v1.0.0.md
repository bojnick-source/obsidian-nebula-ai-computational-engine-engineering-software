**Core formalism:**

Time-independent Schrödinger equation:
    ĤΨ = EΨ,  Ĥ = -ℏ²/2m ∇² + V(r)

Bra-ket notation:
    ⟨φ|Ĥ|ψ⟩ = ∫ φ*(r) Ĥ ψ(r) d³r
Expectation value: ⟨Â⟩ = ⟨ψ|Â|ψ⟩

## Step 1 — Particle in a Box (Quantum Confinement)

```python
import numpy as np
from scipy.constants import hbar, m_e, eV

def infinite_square_well(
    L_nm: float,          # well width (nm)
    m_eff_ratio: float = 1.0,  # effective mass m*/m_e
    n_states: int = 10,
) -> dict:
    """
    Energy levels of 1D infinite square well (particle in a box).
    En = n²π²ℏ²/(2mL²)

    Application: quantum confinement in quantum dots, quantum wells,
    2DEG in GaN HEMT, electron energy levels in nanowires.
    """
    L_m = L_nm * 1e-9
    m   = m_eff_ratio * m_e

    E_n = np.array([
        n**2 * np.pi**2 * hbar**2 / (2 * m * L_m**2) / eV   # eV
        for n in range(1, n_states+1)
    ])

    E1 = E_n[0]   # ground state energy (confinement energy)

    return {
        "E_n_eV": E_n.tolist(),
        "E1_ground_state_eV": round(E1, 4),
        "Delta_E12_eV": round(E_n[1] - E_n[0], 4),   # first optical transition
        "L_nm": L_nm,
        "application": f"Quantum dot: emission wavelength ~ {1240/(E_n[1]-E_n[0]):.0f} nm",
    }
```

## Step 2 — Time-Independent Perturbation Theory

```python
def first_order_perturbation(
    E0_n: float,          # unperturbed energy of state n (eV)
    H_prime_nn: float,    # matrix element ⟨n|H'|n⟩ (eV) — first-order correction
    E0_other: list[float],  # other unperturbed energies
    H_prime_n_others: list[float],  # ⟨n|H'|m⟩ matrix elements
) -> dict:
    """
    First-order energy correction: E_n^(1) = ⟨n|H'|n⟩
    Second-order: E_n^(2) = Σ_{m≠n} |⟨m|H'|n⟩|² / (E_n^(0) - E_m^(0))

    Applications:
    - Stark effect (electric field → level splitting)
    - Zeeman effect (magnetic field → spin splitting)
    - Crystal field splitting (symmetry-reduced environment)
    """
    E1_n = H_prime_nn

    # Second-order correction
    E2_n = sum(
        abs(H_nm)**2 / (E0_n - E0_m)
        for E0_m, H_nm in zip(E0_other, H_prime_n_others)
        if abs(E0_n - E0_m) > 1e-12
    )

    E_total = E0_n + E1_n + E2_n

    return {
        "E0_n_eV": round(E0_n, 6),
        "E1_correction_eV": round(E1_n, 6),
        "E2_correction_eV": round(E2_n, 8),
        "E_corrected_eV": round(E_total, 6),
        "perturbation_valid": abs(E1_n) < abs(E0_n) * 0.1,
        "note": "Perturbation theory valid only if E(1) << E(0); use degenerate PT if E levels degenerate",
    }
```

## Step 3 — Density of States and Fermi-Dirac Statistics

```python
def density_of_states_3d(
    E_eV: np.ndarray,
    E_c_eV: float,        # band edge (conduction band minimum for electrons)
    m_eff_ratio: float,   # effective mass m*/m_e
) -> np.ndarray:
    """
    3D parabolic band DOS:
    g(E) = (1/2π²) × (2m*/ℏ²)^(3/2) × √(E-Ec) for E > Ec

    Units: m⁻³ eV⁻¹
    """
    from scipy.constants import hbar, m_e, eV
    m = m_eff_ratio * m_e
    const = (1/(2*np.pi**2)) * (2*m/hbar**2)**1.5

    dos = np.where(
        E_eV >= E_c_eV,
        const * np.sqrt((E_eV - E_c_eV) * eV) * eV**(-1) * 1e-6,  # → m⁻³/eV
        0.0
    )
    return dos


def fermi_dirac(E_eV: np.ndarray, E_F_eV: float, T_K: float) -> np.ndarray:
    """Fermi-Dirac distribution f(E) = 1/(1+exp((E-EF)/kT))."""
    from scipy.constants import k, eV
    kT = k * T_K / eV
    return 1 / (1 + np.exp((E_eV - E_F_eV) / kT))


def carrier_concentration_3d(
    E_F_eV: float, E_c_eV: float, m_eff_ratio: float, T_K: float,
) -> float:
    """
    Electron concentration n = ∫ g(E) f(E) dE (integrated numerically).
    Returns n in cm⁻³.
    """
    from scipy.integrate import quad
    from scipy.constants import hbar, m_e, eV, k

    m = m_eff_ratio * m_e
    kT_J = k * T_K
    Nc = 2 * (2*np.pi*m*kT_J / hbar**2)**1.5 / (4*np.pi**3)  # effective DOS
    eta = (E_F_eV - E_c_eV) * eV / kT_J   # reduced Fermi energy

    # Fermi-Dirac integral F_{1/2}(η)
    from scipy.special import gamma as gamma_func
    from scipy.integrate import quad
    def F_half(eta_):
        integrand = lambda x: np.sqrt(x) / (1 + np.exp(x - eta_))
        result, _ = quad(integrand, 0, 100)
        return result

    n_m3 = (2/np.sqrt(np.pi)) * Nc * F_half(eta)
    return n_m3 * 1e-6   # cm⁻³
```

## Step 4 — WKB Quantum Tunneling

```python
def wkb_transmission(
    V_barrier_eV: float,    # barrier height above particle energy
    E_particle_eV: float,   # particle kinetic energy
    d_barrier_nm: float,    # barrier width
    m_eff_ratio: float = 1.0,
) -> dict:
    """
    WKB transmission coefficient for rectangular barrier.
    T ≈ exp(-2 ∫ κ(x) dx),  κ = √(2m(V-E)/ℏ²)

    Applications:
    - Fowler-Nordheim tunneling (field emission, MOSFET gate dielectrics)
    - Alpha particle emission
    - STM imaging contrast
    - Josephson junctions (superconducting weak links)
    """
    from scipy.constants import hbar, m_e, eV

    dE = V_barrier_eV - E_particle_eV
    if dE <= 0:
        return {"T_WKB": 1.0, "note": "Classical transmission — no tunneling (E > V)"}

    m = m_eff_ratio * m_e
    kappa = np.sqrt(2 * m * dE * eV) / hbar
    d_m  = d_barrier_nm * 1e-9

    T = np.exp(-2 * kappa * d_m)

    return {
        "T_WKB": round(T, 8),
        "kappa_m_inv": round(kappa, 6),
        "kappa_inv_length_nm": round(1/kappa * 1e9, 3),
        "note": f"T={T:.2e}; tunneling significant when T > 10⁻⁴",
    }
```
