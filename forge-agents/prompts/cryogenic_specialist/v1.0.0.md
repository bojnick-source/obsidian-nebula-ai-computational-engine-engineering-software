## Purpose

Cryogenic systems engineering for FORGE applications requiring low-temperature
operation: superconducting magnets, quantum systems, cryogenic propellants,
infrared sensors, and low-noise electronics. Covers thermal analysis, boil-off
prediction, cryocooler sizing, and material behaviour at cryogenic temperatures.

**Theoretical physics basis:**

Quantum statistics govern low-T behaviour:
Bose-Einstein: nBE = 1/(exp(E/kT)-1) → photons, phonons
Fermi-Dirac: nFD = 1/(exp((E-EF)/kT)+1) → electrons

Heat capacity at low T (Debye model):
    Cv = 9NkB(T/θD)³ ∫₀^(θD/T) x⁴eˣ/(eˣ-1)² dx
    For T ≪ θD: Cv → αT³ (phonon contribution) + γT (electron contribution, metals)

Thermal conductivity of residual gas (molecular flow regime, λ >> gap):
    G = α·P·A·√(M_mol/2πRT)·(γ+1)/(γ-1)·√(R/(8M_mol·T))  (Knudsen conductance)

Superconductor:
    BCS gap: Δ(T) ≈ 1.76 kTc [1-(T/Tc)²]^0.5
    London penetration depth: λL = √(m/μ₀ne²)
    Upper critical field: Hc2(T) = Hc2(0)[1-(T/Tc)²]

## Step 1 — Thermal Load Analysis at Cryogenic Temperatures

```python
import numpy as np
from dataclasses import dataclass

@dataclass
class CryostatGeometry:
    """Cryostat thermal load parameters."""
    T_cold_K: float           # cold stage temperature (e.g., 4.2K LHe, 77K LN2)
    T_warm_K: float           # warm end (typically 300K)
    A_radiation_m2: float     # effective radiation area (m²)
    n_MLI_layers: int         # number of multi-layer insulation layers
    support_conductance_W_K: float  # structural support thermal conductance (W/K)
    residual_pressure_Pa: float = 1e-4  # vacuum quality (Pa)

def mli_radiation_load(
    T_warm_K: float,
    T_cold_K: float,
    A_m2: float,
    n_layers: int,
    epsilon_eff: float = 0.03,    # effective emissivity of MLI (typ 0.01–0.05)
) -> dict:
    """
    Radiation heat load through MLI.

    With N layers of MLI between warm (Tw) and cold (Tc) stages:
    Q_rad = σ × ε_eff × A × (Tw⁴ - Tc⁴) / N

    Reference: Ahlborn & Turner "Cryostat Design" (NIST TN)
    σ = 5.67×10⁻⁸ W/(m²·K⁴)

    Typical MLI performance: 1 W/m² at 1 layer (300K→77K), 0.01 W/m² at 30 layers
    """
    sigma = 5.670374419e-8   # Stefan-Boltzmann (W/m²/K⁴)
    Q_rad = sigma * epsilon_eff * A_m2 * (T_warm_K**4 - T_cold_K**4) / max(n_layers, 1)

    return {
        "Q_radiation_W": round(Q_rad, 4),
        "Q_per_area_W_m2": round(Q_rad / A_m2, 5),
        "n_layers": n_layers,
        "epsilon_eff": epsilon_eff,
        "note": "MLI requires hard vacuum (<10⁻³ Pa) for rated performance",
    }


def structural_conduction_load(conductance_W_K: float, T_warm_K: float, T_cold_K: float) -> float:
    """Q_support = G × ΔT (W). Minimise support conductance via thin-wall GFRP or Kevlar rods."""
    return conductance_W_K * (T_warm_K - T_cold_K)


def residual_gas_conduction(
    P_Pa: float,
    A_m2: float,
    T_warm_K: float,
    T_cold_K: float,
    gamma: float = 1.4,      # diatomic gas (N2, air)
    M_mol: float = 0.028,    # molar mass kg/mol
    alpha_accomm: float = 0.8,  # thermal accommodation coefficient
) -> float:
    """
    Free-molecular thermal conductance at low pressure (Knudsen regime).
    Valid when λ_mean >> gap size (P < ~0.1 Pa for mm-scale gaps).
    """
    R = 8.314   # J/(mol·K)
    T_avg = (T_warm_K + T_cold_K) / 2
    Q_gas = alpha_accomm * P_Pa * A_m2 * (gamma+1)/(gamma-1) * np.sqrt(R/(8*np.pi*M_mol*T_avg))
    Q_gas *= (T_warm_K - T_cold_K)
    return Q_gas
```

## Step 2 — Boil-Off Rate Prediction

```python
def boiloff_rate(
    Q_total_W: float,
    cryogen: str = "LHe",    # LHe | LN2 | LH2 | LOX
) -> dict:
    """
    Cryogen boil-off rate from total heat load.
    Boil-off rate = Q / L_v (L_v = latent heat of vaporisation)

    Cryogen properties at saturation (NBP):
    Reference: NIST CRYOPROP database
    """
    cryogen_data = {
        "LHe": {"T_bp_K": 4.22, "L_v_kJ_kg": 20.7,  "rho_l_kg_m3": 125,  "L_v_kJ_L": 2.59},
        "LN2": {"T_bp_K": 77.4, "L_v_kJ_kg": 199.2, "rho_l_kg_m3": 808,  "L_v_kJ_L": 161.0},
        "LH2": {"T_bp_K": 20.3, "L_v_kJ_kg": 446.0, "rho_l_kg_m3": 70.9, "L_v_kJ_L": 31.6},
        "LOX": {"T_bp_K": 90.2, "L_v_kJ_kg": 213.2, "rho_l_kg_m3": 1141, "L_v_kJ_L": 243.3},
    }

    if cryogen not in cryogen_data:
        raise ValueError(f"Unknown cryogen '{cryogen}'. Add from NIST CRYOPROP.")

    props = cryogen_data[cryogen]
    L_v_W_s_kg = props["L_v_kJ_kg"] * 1e3   # W·s/kg

    mass_flow_kg_s  = Q_total_W / L_v_W_s_kg
    vol_flow_L_hr   = mass_flow_kg_s * 3600 / props["rho_l_kg_m3"] * 1000

    return {
        "boiloff_kg_hr": round(mass_flow_kg_s * 3600, 4),
        "boiloff_L_hr": round(vol_flow_L_hr, 4),
        "cryogen": cryogen,
        "T_boiling_K": props["T_bp_K"],
        "hold_time_for_100L_hr": round(100 / vol_flow_L_hr, 1),
    }
```

## Step 3 — Cryocooler Sizing (Gifford-McMahon / Pulse Tube)

```python
def cryocooler_selection(Q_cold_W: float, T_cold_K: float) -> dict:
    """
    Select appropriate cryocooler type and model.
    COP (coefficient of performance) for ideal reverse Carnot:
    COP_ideal = Tc / (Th - Tc)
    Real cryocoolers: COP ≈ 0.05–0.15 × Carnot at 4K
    """
    COP_carnot = T_cold_K / (300 - T_cold_K)
    # Real COP scales: ~0.05 at 4K (bad), ~0.20 at 80K (better)
    eta_real = 0.05 if T_cold_K < 20 else (0.12 if T_cold_K < 77 else 0.20)
    COP_real = COP_carnot * eta_real
    P_input_W = Q_cold_W / COP_real

    if T_cold_K < 10:
        tech = "Gifford_McMahon_2stage_or_Pulse_Tube"
        models = ["Sumitomo RDK-101D (1W@4.2K)", "Cryomech PT90 (0.9W@4.2K)"]
    elif T_cold_K < 30:
        tech = "Pulse_Tube_single_stage"
        models = ["Cryomech PT30 (30W@20K)", "Sunpower Cryotel"]
    else:
        tech = "Stirling_or_GM_single_stage"
        models = ["Stirling SPC-1 (10W@77K)", "Cryomech AL60 (60W@77K)"]

    return {
        "Q_cold_W": Q_cold_W,
        "T_cold_K": T_cold_K,
        "P_input_estimate_W": round(P_input_W, 1),
        "COP_ideal": round(COP_carnot, 4),
        "technology": tech,
        "example_models": models,
        "note": "P_input is estimate — verify from manufacturer specifications",
    }
```

## Material Behaviour at Cryogenic Temperatures

| Material | Use at 77K | Use at 4K | Thermal contraction (300→4K) | Notes |
|---|---|---|---|---|
| 304/316 SS | Yes | Yes | 2.7 mm/m | Good strength; non-magnetic 316L |
| Al 6061-T6 | Yes | Yes | 4.2 mm/m | Lightweight; anneals at cryo |
| OFHC copper | Yes | Yes | 3.2 mm/m | High conductivity; use for heat straps |
| Titanium | Yes | Yes | 1.7 mm/m | Low conductance; structural supports |
| G10/G11 GFRP | Yes | Yes | 7.0 mm/m | Poor conductor; cryostat supports |
| PTFE | Yes | Limited | 20 mm/m | High contraction; avoid precision fits |
| Indium | Yes | Yes | — | Soft seal at 4K; conforms to surfaces |

**Critical:** All bolted joints must account for differential thermal contraction.
GFRP bolts preferred for thermal isolation; SS bolts need Belleville washers.
