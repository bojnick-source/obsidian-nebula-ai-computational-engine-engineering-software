## Purpose

Solid state physics provides the microscopic foundation for all crystalline material
properties in FORGE. Connects crystal structure and electronic band structure to
mechanical, electrical, thermal, and optical properties.

**Core formalism:**

Bloch's theorem (periodic potential):
    ψ_nk(r) = u_nk(r) e^(ik·r)
where u_nk has the lattice periodicity: u_nk(r+R) = u_nk(r)

Brillouin zone: reciprocal lattice primitive cell
    k ∈ 1st BZ → k_max = π/a for simple cubic (a = lattice constant)

Band gap:
    E_g = E_CBM - E_VBM  (direct: k_VBM = k_CBM; indirect: k_VBM ≠ k_CBM)

Drude model conductivity:
    σ = ne²τ/m = neμ,  μ = eτ/m* (carrier mobility)
    τ = mean free time between collisions

Peierls-Nabarro lattice friction (dislocation motion):
    τ_PN = (2G/1-ν) exp(-2πd/b(1-ν))  (G=shear modulus, b=Burgers vector, d=spacing)

## Step 1 — Crystal Structure and X-Ray Diffraction

```python
import numpy as np
from itertools import product

def bragg_diffraction(
    d_spacing_angstrom: float,   # interplanar spacing (from crystal structure)
    lambda_Angstrom: float = 1.5406,  # Cu Kα X-ray wavelength
    order_n: int = 1,
) -> dict:
    """
    Bragg's law: nλ = 2d sin(θ)
    θ = arcsin(nλ / 2d)

    Returns 2θ diffraction angle (degrees).
    Used to identify crystal phases, measure strain (Δd/d → ε).
    """
    sin_theta = order_n * lambda_Angstrom / (2 * d_spacing_angstrom)
    if sin_theta > 1.0:
        return {"status": "no_diffraction", "reason": f"nλ/2d = {sin_theta:.3f} > 1"}

    theta_deg = np.degrees(np.arcsin(sin_theta))
    two_theta = 2 * theta_deg

    return {
        "theta_deg": round(theta_deg, 3),
        "two_theta_deg": round(two_theta, 3),
        "d_angstrom": d_spacing_angstrom,
        "n": order_n,
    }


def fcc_structure_factor(h: int, k: int, l: int) -> float:
    """
    FCC structure factor: F = f × Σ exp(2πi(hx+ky+lz))
    FCC basis: (0,0,0), (½,½,0), (½,0,½), (0,½,½)
    F≠0 only when h,k,l are all odd or all even.
    """
    basis = [(0,0,0), (0.5,0.5,0), (0.5,0,0.5), (0,0.5,0.5)]
    F = sum(np.exp(2j * np.pi * (h*x + k*y + l*z)) for x,y,z in basis)
    return abs(F)   # |F|: 4 if mixed (forbidden), 0 if all-same parity
```

## Step 2 — Electronic Band Structure (Tight-Binding)

```python
def tight_binding_1d(
    t_eV: float,        # hopping integral (coupling energy, negative for bonding)
    epsilon_eV: float,  # on-site energy
    k_pts: int = 200,
) -> dict:
    """
    Nearest-neighbour tight-binding band structure for 1D chain.
    E(k) = ε₀ + 2t cos(ka)  (a = lattice constant, normalised to 1)

    Physical meaning:
    - t < 0: bonding orbital → filled band → insulator/semiconductor
    - t > 0: anti-bonding
    - Bandwidth W = 4|t| → broader bands = more delocalised (metallic)

    Application: graphene (2D: E = ±ℏvF|k|, vF~10⁶m/s from t=-2.7eV)
    """
    k = np.linspace(-np.pi, np.pi, k_pts)   # 1st BZ
    E = epsilon_eV + 2 * t_eV * np.cos(k)

    bandwidth  = 4 * abs(t_eV)
    E_min, E_max = E.min(), E.max()

    return {
        "k_pi_units": k / np.pi,
        "E_k_eV": E,
        "bandwidth_eV": round(bandwidth, 4),
        "E_min_eV": round(E_min, 4),
        "E_max_eV": round(E_max, 4),
        "effective_mass_m_e": round(-hbar**2 / (2 * t_eV * 1.602e-19 * 1e-20), 3),
    }
```

## Step 3 — Phonon Dispersion (Diatomic Chain)

```python
def diatomic_chain_phonons(
    K_N_m: float,     # spring constant (N/m)
    M1_kg: float,     # mass of atom 1
    M2_kg: float,     # mass of atom 2
    k_pts: int = 200,
) -> dict:
    """
    Acoustic and optical phonon branches for 1D diatomic chain.
    ω² = K(1/M1 + 1/M2) ± K√((1/M1+1/M2)² - 4sin²(ka)/M1M2)

    Gap between acoustic and optical branches:
    - Acoustic: ω→0 as k→0 (sound waves)
    - Optical: ω → √(2K(1/M1+1/M2)) at k=0 (IR active)

    Thermal conductivity κ ~ Cv × v_group × l_mfp
    v_group = dω/dk (group velocity) → determines phonon heat transport
    """
    a = 1.0   # unit cell (normalised)
    k = np.linspace(-np.pi/(2*a), np.pi/(2*a), k_pts)

    A = K * (1/M1 + 1/M2)
    B = K**2 * 4 * np.sin(k*a)**2 / (M1 * M2)

    omega_sq_acoustic = A - np.sqrt(np.maximum(A**2 - B, 0))
    omega_sq_optical  = A + np.sqrt(np.maximum(A**2 - B, 0))

    omega_acoustic = np.sqrt(np.abs(omega_sq_acoustic))
    omega_optical  = np.sqrt(omega_sq_optical)

    v_sound = float(omega_acoustic[k_pts//2+1] / abs(k[k_pts//2+1]) if k[k_pts//2+1] != 0 else 0)

    return {
        "omega_acoustic_THz": omega_acoustic / 1e12,
        "omega_optical_THz": omega_optical / 1e12,
        "omega_optical_at_k0_THz": round(float(omega_optical[k_pts//2]) / 1e12, 4),
        "v_sound_m_s": round(v_sound, 1),
        "phonon_gap_THz": round(float((omega_optical[0] - omega_acoustic[-1])) / 1e12, 4),
    }
```

## Step 4 — Dislocation and Plastic Deformation

```python
def burgers_vector_energy(
    b_m: float,         # Burgers vector magnitude (m) — lattice parameter or fraction
    G_GPa: float,       # shear modulus
    nu: float,          # Poisson's ratio
    r_inner_m: float,   # inner cutoff (core radius, ~b)
    r_outer_m: float,   # outer cutoff (grain size or crystal size)
    dislocation_type: str = "edge",   # edge | screw
) -> dict:
    """
    Elastic strain energy per unit length of dislocation.
    Edge:   E_line = Gb²/4π(1-ν) × ln(r_outer/r_inner)
    Screw:  E_line = Gb²/4π      × ln(r_outer/r_inner)

    Frank's rule: dislocation reaction b₁→b₂+b₃ energetically favourable
    if b₁² > b₂² + b₃² (|b|² decreases = energy decreases).

    Application: work hardening (Taylor relation τ = αGb√ρ, ρ=disloc density)
    """
    G_Pa = G_GPa * 1e9
    ratio = np.log(r_outer_m / r_inner_m)

    if dislocation_type == "edge":
        E_line = G_Pa * b_m**2 / (4 * np.pi * (1 - nu)) * ratio
    else:  # screw
        E_line = G_Pa * b_m**2 / (4 * np.pi) * ratio

    return {
        "E_line_J_m": round(E_line, 6),
        "E_line_eV_nm": round(E_line * 1e9 / 1.602e-19, 3),
        "type": dislocation_type,
        "b_nm": round(b_m * 1e9, 4),
    }
```
