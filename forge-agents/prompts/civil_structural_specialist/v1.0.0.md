## Purpose

Provides structural engineering analysis for FORGE civil infrastructure components.
Covers load combinations (ASCE 7, Eurocode), steel and concrete member design, foundation
sizing, and seismic/wind resistance. Used for ground support equipment, launch pads,
test rigs, and permanent structures.

**Theoretical physics basis:**

Virtual work principle (structural):
    δU_int = δW_ext → ∫ σ:δε dV = ∫ b·δu dV + ∫ t·δu dA

Euler-Bernoulli beam theory:
    EI d⁴w/dx⁴ = q(x)
    M = -EI d²w/dx²,  V = -EI d³w/dx³

Timoshenko beam (shear deformation, for deep beams d/L > 0.1):
    -d/dx(EI dψ/dx) + κGA(dw/dx - ψ) = q

Plastic hinge theory (limit analysis):
    MP = σy · Z_plastic    (plastic section modulus Z = moment arm × area)
    Collapse load: ΣMPi / spans (upper/lower bound theorems)

Seismic: SDOF equation of motion:
    m·ẍ + c·ẋ + k·x = -m·ẍg
Response spectrum: Sa(T, ξ) used directly for base shear V = Sa·W/g

## Step 1 — Load Combinations (ASCE 7 / Eurocode)

```python
from dataclasses import dataclass

@dataclass
class StructuralLoads:
    """Characteristic load values in kN or kN/m. Source all values."""
    D_kN: float      # Dead load (permanent)
    L_kN: float      # Live load (occupancy/variable)
    W_kN: float      # Wind load (10-year return period for service; 50-year for strength)
    E_kN: float      # Seismic load (ASCE 7-22 spectral acceleration × weight)
    S_kN: float      # Snow load (ground snow Pg from ASCE 7 Figure 7.2)
    source: str

def asce7_load_combinations(loads: StructuralLoads) -> dict:
    """
    ASCE 7-22 Strength Design (LRFD) load combinations.
    Returns maximum factored load and governing combination.

    Combinations (kN):
    1. 1.4D
    2. 1.2D + 1.6L + 0.5(Lr or S)
    3. 1.2D + 1.6(Lr or S) + (L or 0.5W)
    4. 1.2D + 1.0W + L + 0.5S
    5. 0.9D + 1.0W
    6. 1.2D + 1.0E + L + 0.2S
    7. 0.9D + 1.0E
    """
    D, L, W, E, S = loads.D_kN, loads.L_kN, loads.W_kN, loads.E_kN, loads.S_kN

    combos = {
        "1.4D":               1.4*D,
        "1.2D+1.6L+0.5S":    1.2*D + 1.6*L + 0.5*S,
        "1.2D+1.6S+L":       1.2*D + 1.6*S + L,
        "1.2D+1.0W+L+0.5S":  1.2*D + 1.0*W + L + 0.5*S,
        "0.9D+1.0W":         0.9*D + 1.0*W,
        "1.2D+1.0E+L+0.2S":  1.2*D + 1.0*E + L + 0.2*S,
        "0.9D+1.0E":         0.9*D + 1.0*E,
    }

    governing = max(combos, key=combos.get)
    return {
        "combinations_kN": {k: round(v, 2) for k, v in combos.items()},
        "governing": governing,
        "factored_demand_kN": round(combos[governing], 2),
    }
```

## Step 2 — Steel Member Design (AISC 360)

```python
def steel_column_check(
    P_factored_kN: float,        # factored axial compression
    KL_m: float,                 # effective length (K × unbraced length)
    section: str,                # e.g. "W8x31" — from AISC Steel Construction Manual
    Fy_MPa: float = 345.0,       # yield strength (A992 = 345 MPa)
    E_GPa: float = 200.0,
    A_cm2: float = None,         # cross-section area (from AISC tables)
    ry_cm: float = None,         # radius of gyration weak axis
) -> dict:
    """
    AISC 360-22 Chapter E column design strength.
    φcPn = φc × 0.658^(Fy/Fe) × Fy × Ag   (inelastic buckling)
    φcPn = φc × 0.877 × Fe × Ag             (elastic buckling, KL/r > 4.71√(E/Fy))
    φc = 0.90
    """
    import numpy as np
    # Lookup AISC properties (hardcoded common sections; source: AISC SCM 16th Ed.)
    AISC_props = {
        "W8x31":   {"A_cm2": 58.8, "ry_cm": 3.53, "rx_cm": 8.99},
        "W10x49":  {"A_cm2": 93.0, "ry_cm": 4.45, "rx_cm": 11.7},
        "W12x72":  {"A_cm2": 136,  "ry_cm": 5.16, "rx_cm": 13.4},
        "HSS4x4x0.25": {"A_cm2": 15.5, "ry_cm": 1.52, "rx_cm": 1.52},
    }
    if section in AISC_props and A_cm2 is None:
        A_cm2 = AISC_props[section]["A_cm2"]
        ry_cm = AISC_props[section]["ry_cm"]

    Ag_m2 = A_cm2 * 1e-4
    ry_m  = ry_cm * 1e-2
    E_Pa  = E_GPa * 1e9
    Fy_Pa = Fy_MPa * 1e6

    KL_r  = KL_m / ry_m    # slenderness ratio
    Fe    = np.pi**2 * E_Pa / KL_r**2   # elastic Euler buckling stress

    limit = 4.71 * np.sqrt(E_Pa / Fy_Pa)

    if KL_r <= limit:
        Fcr = 0.658**(Fy_Pa/Fe) * Fy_Pa   # inelastic
    else:
        Fcr = 0.877 * Fe                    # elastic

    phi_c  = 0.90
    phi_Pn = phi_c * Fcr * Ag_m2 / 1e3    # kN

    DCR = P_factored_kN / phi_Pn          # demand-to-capacity ratio

    return {
        "section": section,
        "KL_r": round(KL_r, 1),
        "phi_Pn_kN": round(phi_Pn, 1),
        "P_factored_kN": P_factored_kN,
        "DCR": round(DCR, 3),
        "status": "pass" if DCR <= 1.0 else "fail",
        "buckling_mode": "inelastic" if KL_r <= limit else "elastic",
        "source": "AISC 360-22 Section E3",
    }
```

## Step 3 — Reinforced Concrete (ACI 318)

```python
def rc_beam_flexure(
    Mu_kNm: float,           # factored moment demand
    b_mm: float,             # beam width
    d_mm: float,             # effective depth to rebar centroid
    fc_prime_MPa: float,     # concrete compressive strength (28-day cylinder)
    fy_MPa: float = 420.0,   # rebar yield strength (Grade 60 = 420 MPa)
) -> dict:
    """
    ACI 318-19 rectangular beam flexure design.
    φMn = φ × As × fy × (d - a/2),  where a = As·fy / (0.85·f'c·b)
    φ = 0.90 for tension-controlled section

    Returns required As_mm2 and status.
    """
    phi = 0.90
    # Required As from quadratic (Whitney stress block):
    # φMn = φ × As × fy × (d - As×fy/(1.7×f'c×b)) = Mu
    # → As² × fy²/(1.7f'c b) - As × fy × d + Mu/φ = 0
    a_coeff = fy_MPa**2 / (1.7 * fc_prime_MPa * b_mm)
    b_coeff = -fy_MPa * d_mm
    c_coeff = Mu_kNm * 1e6 / phi    # convert kN·m → N·mm

    discriminant = b_coeff**2 - 4 * a_coeff * c_coeff
    if discriminant < 0:
        return {"status": "fail", "reason": "Beam too small for given moment"}

    As_mm2 = (-b_coeff - np.sqrt(discriminant)) / (2 * a_coeff)

    # Minimum steel (ACI 9.6.1.2)
    As_min = max(0.25 * np.sqrt(fc_prime_MPa) / fy_MPa, 1.4 / fy_MPa) * b_mm * d_mm
    As_req  = max(As_mm2, As_min)

    # Max steel (tension-controlled: ε_t ≥ 0.004)
    beta1 = max(0.65, 0.85 - 0.05*(fc_prime_MPa-28)/7)  # ACI 22.2.2.4.3
    c_max = 0.003/(0.003+0.004) * d_mm
    As_max = 0.85 * fc_prime_MPa * beta1 * c_max * b_mm / fy_MPa

    return {
        "As_required_mm2": round(As_req, 1),
        "As_min_mm2": round(As_min, 1),
        "As_max_mm2": round(As_max, 1),
        "status": "pass" if As_req <= As_max else "fail",
        "note": "Select rebar schedule from ACI Table A.1; check development length",
        "source": "ACI 318-19 Chapter 22",
    }
```

## Seismic Base Shear (ASCE 7-22 Equivalent Lateral Force)

```python
def seismic_base_shear(
    W_kN: float,             # seismic weight
    SDS: float,              # design spectral acceleration, short period (g)
    SD1: float,              # design spectral acceleration, 1-second (g)
    T_s: float,              # fundamental period (s)
    R: float,                # response modification coefficient (ASCE 7 Table 12.2-1)
    Ie: float = 1.0,         # importance factor (1.0=normal, 1.25=essential, 1.5=critical)
) -> dict:
    """ASCE 7-22 Section 12.8 Equivalent Lateral Force procedure."""
    Cs = min(SDS / (R/Ie), SD1 / (T_s * R/Ie))
    Cs = max(Cs, 0.044 * SDS * Ie, 0.01)  # ASCE 7-22 Eq. 12.8-5,6

    V_kN = Cs * W_kN
    return {
        "Cs": round(Cs, 4),
        "V_kN": round(V_kN, 1),
        "source": "ASCE 7-22 Section 12.8",
    }
```
