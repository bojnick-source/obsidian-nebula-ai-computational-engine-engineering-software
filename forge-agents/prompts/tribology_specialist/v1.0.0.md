## Purpose

Analyses contact mechanics, friction, wear, and lubrication for FORGE components.
Critical for bearings, actuators, joints, seals, and any surface-to-surface interface.
Prevents galling, fretting fatigue, and premature wear in moving systems.

**Theoretical physics basis:**

Hertzian contact mechanics (Hertz 1882):
Sphere-on-flat: a = (3FR/4E*)^(1/3), E* = [(1-ν₁²)/E₁ + (1-ν₂²)/E₂]^(-1)
Maximum contact pressure: p₀ = 3F/(2πa²)

Adhesive friction (Bowden & Tabor 1950):
    μ = τ_s / H    (shear strength of interface / hardness)

Archard wear law (Archard 1953):
    V = k · F · s / H
where V = wear volume, k = dimensionless wear coefficient, s = sliding distance, H = hardness

EHL film thickness (Hamrock-Dowson 1977, point contact):
    h_min = 3.63 · R · (η₀V/E*R)^0.68 · (αE*)^0.49 · (1 - e^(-0.68κ)) / (F/(E*R²))^0.073

Stribeck curve regimes:
    Boundary:    λ = h/σ < 1 → metal-to-metal contact, high wear
    Mixed:       1 < λ < 3 → partial film, moderate wear
    Full film:   λ > 3 → hydrodynamic/EHL, near-zero wear

## Step 1 — Hertzian Contact Analysis

```python
import numpy as np
from dataclasses import dataclass

@dataclass
class ContactBody:
    """Geometric and material properties of one contact body."""
    E_GPa: float         # Young's modulus
    nu: float            # Poisson's ratio
    H_GPa: float         # Hardness (Vickers, converted to GPa: HV × 9.81e-3)
    R_mm: float          # Effective radius (set to 1e9 for flat)
    roughness_Ra_um: float  # Surface roughness Ra

def hertz_contact(
    body1: ContactBody,
    body2: ContactBody,
    normal_force_N: float,
    geometry: str = "sphere_flat",  # sphere_flat | cylinder_flat | sphere_sphere
) -> dict:
    """
    Hertzian contact mechanics.
    Returns: contact_radius_mm, max_pressure_MPa, max_shear_depth_mm,
             lambda_ratio (film thickness to roughness)
    """
    E1, E2 = body1.E_GPa * 1e9, body2.E_GPa * 1e9  # Pa
    nu1, nu2 = body1.nu, body2.nu

    # Combined modulus
    E_star = 1 / ((1 - nu1**2)/E1 + (1 - nu2**2)/E2)  # Pa

    # Effective radius
    R1, R2 = body1.R_mm * 1e-3, body2.R_mm * 1e-3  # m
    if geometry == "sphere_flat":
        R_eff = R1   # R2 → ∞
    elif geometry == "sphere_sphere":
        R_eff = 1 / (1/R1 + 1/R2)
    elif geometry == "cylinder_flat":
        # 2D: contact half-width b = 2√(FR/πE*L)
        R_eff = R1

    F = normal_force_N

    if geometry in ["sphere_flat", "sphere_sphere"]:
        a = (3 * F * R_eff / (4 * E_star))**(1/3)   # m
        p0 = 3 * F / (2 * np.pi * a**2)              # Pa
        max_shear_depth = 0.48 * a                    # depth of max shear stress

    elif geometry == "cylinder_flat":
        L = 1.0  # per unit length
        b = np.sqrt(4 * F * R_eff / (np.pi * E_star * L))  # half-width
        p0 = 2 * F / (np.pi * b * L)
        a  = b
        max_shear_depth = 0.786 * b

    # Combined roughness
    sigma_combined = np.sqrt(body1.roughness_Ra_um**2 + body2.roughness_Ra_um**2) * 1.25  # σ=1.25Ra

    # Yield check (subsurface yield when p0 > 1.1 × hardness of softer material)
    H_min = min(body1.H_GPa, body2.H_GPa) * 1e9   # Pa
    yield_flag = p0 > 1.1 * H_min

    return {
        "contact_radius_mm": round(a * 1e3, 4),
        "max_pressure_MPa": round(p0 * 1e-6, 2),
        "max_shear_depth_mm": round(max_shear_depth * 1e3, 4),
        "E_star_GPa": round(E_star * 1e-9, 2),
        "subsurface_yield": yield_flag,
        "subsurface_yield_note": "Plastic deformation expected — Archard wear accelerated" if yield_flag else "Elastic contact",
    }
```

## Step 2 — Lubrication and Film Thickness

```python
def ehl_film_thickness(
    R_eff_m: float,
    F_N: float,
    E_star_Pa: float,
    V_ms: float,             # entrainment velocity (m/s)
    eta0_Pa_s: float,        # dynamic viscosity at p=0 (e.g., 0.05 Pa·s for ISO VG 46)
    alpha_Pa_inv: float,     # pressure-viscosity coefficient (typ. 1.5–2.5×10⁻⁸ Pa⁻¹)
    kappa: float = 1.0,      # ellipticity parameter (1=circular, >1=elliptical)
) -> dict:
    """
    Hamrock-Dowson minimum film thickness for point contact EHL.
    h_min = 3.63 R (η₀V/E*R)^0.68 (αE*)^0.49 (1 - e^(-0.68κ)) / (F/E*R²)^0.073

    Reference: Hamrock & Dowson (1977) ASME J. Lubr. Technol.
    """
    W = F_N / (E_star_Pa * R_eff_m**2)   # dimensionless load
    U = eta0_Pa_s * V_ms / (E_star_Pa * R_eff_m)  # dimensionless speed
    G = alpha_Pa_inv * E_star_Pa          # dimensionless materials

    # Hamrock-Dowson (1977)
    h_min = 3.63 * R_eff_m * U**0.68 * G**0.49 * (1 - np.exp(-0.68 * kappa)) * W**(-0.073)

    return {
        "h_min_um": round(h_min * 1e6, 3),
        "h_min_m": h_min,
        "dimensionless_load_W": round(W, 6),
        "dimensionless_speed_U": round(U, 8),
        "dimensionless_material_G": round(G, 2),
    }


def stribeck_regime(
    h_min_um: float,
    surface1_Ra_um: float,
    surface2_Ra_um: float,
) -> dict:
    """
    Determine Stribeck lubrication regime from film thickness ratio λ.
    λ = h_min / σ,  σ = √(Ra₁² + Ra₂²)
    """
    sigma_um = np.sqrt(surface1_Ra_um**2 + surface2_Ra_um**2)
    lam = h_min_um / max(sigma_um, 0.001)

    if lam < 1:
        regime = "boundary"
        wear_rate = "high"
        description = "Metal-to-metal contact. Boundary additives (ZDDP, MoS₂) required."
    elif lam < 3:
        regime = "mixed"
        wear_rate = "moderate"
        description = "Partial film. Reduce roughness or increase speed/viscosity."
    else:
        regime = "full_film"
        wear_rate = "low"
        description = "Full EHL or hydrodynamic film. Wear negligible if λ maintained."

    return {
        "lambda_ratio": round(lam, 2),
        "regime": regime,
        "wear_rate": wear_rate,
        "description": description,
        "sigma_combined_Ra_um": round(sigma_um, 3),
    }
```

## Step 3 — Archard Wear Life

```python
def archard_wear_life(
    normal_force_N: float,
    sliding_distance_m: float,
    hardness_softer_GPa: float,
    wear_coefficient_k: float,     # Archard k, dimensionless (cite source: Rabinowicz tables)
    wear_volume_limit_mm3: float,  # acceptable wear volume (from clearance budget)
) -> dict:
    """
    Archard adhesive wear prediction.
    V = k · F · s / H

    Typical k values (Rabinowicz 1965 / Ludema 1996):
    - Like metals, dry: 10⁻³ to 10⁻²  (aggressive)
    - Unlike metals, dry: 10⁻⁴ to 10⁻³
    - Metal on metal, lubricated: 10⁻⁵ to 10⁻⁴
    - PTFE, DLC coating: 10⁻⁶ to 10⁻⁷ (excellent)

    Source for k MUST be cited.
    """
    H_Pa = hardness_softer_GPa * 1e9
    V_m3 = wear_coefficient_k * normal_force_N * sliding_distance_m / H_Pa
    V_mm3 = V_m3 * 1e9

    life_fraction = V_mm3 / wear_volume_limit_mm3

    return {
        "wear_volume_mm3": round(V_mm3, 4),
        "wear_volume_limit_mm3": wear_volume_limit_mm3,
        "life_fraction": round(life_fraction, 3),
        "status": "pass" if life_fraction < 0.8 else "fail",
        "k_used": wear_coefficient_k,
        "note": "Archard wear is order-of-magnitude estimate. Validate with tribometer test (ASTM G99).",
    }
```
