## Purpose

Numerical analysis for FORGE computational pipelines. Provides stable, accurate
algorithms for ODE integration, numerical differentiation/integration, root
finding, and interpolation. Identifies stability limits (CFL, step size) before
they manifest as diverged simulations.

**Core error analysis:**

Global truncation error: O(h^p) for p-th order method
Round-off error: O(ε_machine/h) for finite differences → total error minimised at h* ~ (ε_mach)^(1/(p+1))
Stability: method is A-stable if region of absolute stability contains entire left half-plane

Lax equivalence theorem (for well-posed linear PDE):
    Consistency + Stability ↔ Convergence

## Step 1 — ODE Integration

```python
import numpy as np
from typing import Callable

def runge_kutta_4(
    f: Callable,           # f(t, y) → dy/dt
    t_span: tuple,         # (t0, tf)
    y0: np.ndarray,
    n_steps: int = 1000,
    adaptive: bool = False,
    rtol: float = 1e-6,
    atol: float = 1e-9,
) -> dict:
    """
    4th-order Runge-Kutta (RK4) fixed-step integrator.

    Error: O(h⁵) local, O(h⁴) global.
    Stability: explicit RK4 is stable for h ≤ 2.8/|λ_max| (Euler);
    for stiff problems (|λ_max|/|λ_min| >> 1), use implicit methods.

    Butcher tableau (RK4):
    c | a
    0 |
    ½ | ½
    ½ | 0  ½
    1 | 0  0  1
    --+--------
    b | 1/6 1/3 1/3 1/6
    """
    t0, tf = t_span
    h  = (tf - t0) / n_steps
    t  = np.linspace(t0, tf, n_steps + 1)
    y  = np.zeros((n_steps + 1, len(y0)))
    y[0] = y0

    for i in range(n_steps):
        ti, yi = t[i], y[i]
        k1 = f(ti,       yi)
        k2 = f(ti + h/2, yi + h*k1/2)
        k3 = f(ti + h/2, yi + h*k2/2)
        k4 = f(ti + h,   yi + h*k3)
        y[i+1] = yi + h/6 * (k1 + 2*k2 + 2*k3 + k4)

    return {"t": t, "y": y, "h": h, "n_steps": n_steps}


def ode_stability_analysis(
    f: Callable,    # f(t, y)
    t0: float,
    y0: np.ndarray,
) -> dict:
    """
    Estimate stiffness ratio: λ_max/λ_min of Jacobian J=∂f/∂y at (t0,y0).
    Stiff if ratio > 100 → use implicit BDF or Radau.
    """
    eps = 1e-7
    n   = len(y0)
    J   = np.zeros((n, n))

    f0 = f(t0, y0)
    for j in range(n):
        y_p = y0.copy(); y_p[j] += eps
        J[:, j] = (f(t0, y_p) - f0) / eps

    eigenvalues = np.linalg.eigvals(J)
    stiff_ratio = abs(eigenvalues.real).max() / max(abs(eigenvalues.real).min(), 1e-15)

    return {
        "eigenvalues": eigenvalues,
        "stiffness_ratio": stiff_ratio,
        "is_stiff": stiff_ratio > 100,
        "recommendation": (
            "Use scipy.integrate.solve_ivp method='BDF' or 'Radau' for stiff systems"
            if stiff_ratio > 100 else
            "RK4 or DOP853 (Dormand-Prince) suitable"
        ),
    }
```

## Step 2 — Finite Difference Differentiation

```python
def finite_difference(
    f: Callable,
    x: float,
    h: float = 1e-5,
    order: int = 1,    # derivative order
    accuracy: int = 2, # 2=O(h²), 4=O(h⁴), 6=O(h⁶)
) -> dict:
    """
    Finite difference approximations with error analysis.

    Central differences (1st derivative):
    O(h²): f'(x) ≈ (f(x+h) - f(x-h)) / 2h
    O(h⁴): f'(x) ≈ (-f(x+2h) + 8f(x+h) - 8f(x-h) + f(x-2h)) / 12h

    Optimal step size:
    h* = (ε_mach)^(1/3) × |f|/|f'''| for 1st derivative O(h²)
    Too small: round-off dominates; too large: truncation dominates.
    """
    eps_mach = np.finfo(float).eps   # ~2.22×10⁻¹⁶ for float64

    if order == 1 and accuracy == 2:
        df = (f(x + h) - f(x - h)) / (2 * h)
        err_trunc = h**2 / 6  # coefficient of leading error term (≈)
    elif order == 1 and accuracy == 4:
        df = (-f(x+2*h) + 8*f(x+h) - 8*f(x-h) + f(x-2*h)) / (12*h)
        err_trunc = h**4 / 30
    elif order == 2 and accuracy == 2:
        df = (f(x+h) - 2*f(x) + f(x-h)) / h**2
        err_trunc = h**2 / 12
    else:
        df = (f(x + h) - f(x)) / h   # forward difference O(h)
        err_trunc = h / 2

    # Optimal h for this method
    h_opt = eps_mach**(1/(order + accuracy)) * abs(f(x) + 1e-300)

    return {
        "derivative": df,
        "h_used": h,
        "h_optimal": round(h_opt, 2),
        "truncation_error_approx": err_trunc,
        "roundoff_error_approx": eps_mach / h,
        "note": f"Use h ≈ {h_opt:.1e} for minimum total error",
    }
```

## Step 3 — Numerical Integration (Quadrature)

```python
from scipy.integrate import quad, dblquad, fixed_quad

def adaptive_quadrature(
    f: Callable,
    a: float, b: float,
    epsabs: float = 1.49e-8,
    epsrel: float = 1.49e-8,
    limit: int = 200,
) -> dict:
    """
    Adaptive Gauss-Kronrod quadrature (scipy quad = QUADPACK DQAGSE).
    Automatic subdivision and error control.

    For smooth functions: Gauss-Legendre achieves exponential convergence.
    For oscillatory: use weight functions (Gauss-Laguerre for ∫₀^∞ e^(-x) f dx).
    For singularities: specify singular points in 'points' argument.
    """
    result, error, info = quad(f, a, b, epsabs=epsabs, epsrel=epsrel,
                                limit=limit, full_output=True)
    return {
        "result": result,
        "absolute_error": error,
        "n_evaluations": info["neval"],
        "converged": error < epsabs + epsrel * abs(result),
    }


def gauss_legendre(
    f: Callable, a: float, b: float, n_points: int = 10,
) -> float:
    """
    Fixed-order Gauss-Legendre quadrature.
    Exact for polynomials of degree ≤ 2n-1.
    n_points=5: exact for degree 9 polynomial.
    Preferred over rectangle/trapezoid for smooth integrands.
    """
    xi, wi = np.polynomial.legendre.leggauss(n_points)
    # Map from [-1,1] to [a,b]
    xp = 0.5*(b-a)*xi + 0.5*(b+a)
    return 0.5*(b-a) * np.dot(wi, f(xp))
```

## Step 4 — Root Finding

```python
def newton_raphson(
    f: Callable, df: Callable,
    x0: float,
    tol: float = 1e-12, max_iter: int = 100,
) -> dict:
    """
    Newton-Raphson: x_{n+1} = x_n - f(x_n)/f'(x_n)
    Convergence: quadratic when f'(x*)≠0 — doubles correct digits each iteration.
    Fails: f'(x_n) → 0 (near inflection); oscillates for complex roots.
    """
    x = x0
    for i in range(max_iter):
        fx   = f(x)
        if abs(fx) < tol:
            return {"root": x, "converged": True, "iterations": i, "f_at_root": fx}
        dfx  = df(x)
        if abs(dfx) < 1e-15:
            return {"converged": False, "reason": "Derivative ≈ 0 (near inflection point)"}
        x -= fx / dfx

    return {"root": x, "converged": False, "iterations": max_iter}
```

## Stability Conditions (CFL and Time-Step)

| Equation type | Method | Stability condition |
|---|---|---|
| Advection (hyperbolic) | FTCS | CFL = u·Δt/Δx ≤ 1 |
| Diffusion (parabolic) | FTCS | r = α·Δt/Δx² ≤ 0.5 |
| Wave equation | Leapfrog | CFL ≤ 1/√d (d=dimensions) |
| Stiff ODE | Explicit Euler | hλ ∈ stability region (|1+hλ|<1) |
| Stiff ODE | BDF-2 | A-stable → any h |

Flag `[STABILITY WARN: CFL > 1]` when explicit time-stepping violates CFL.
