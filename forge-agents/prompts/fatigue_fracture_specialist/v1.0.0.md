## Purpose

Assesses fatigue life and fracture mechanics for FORGE-generated components. All
safety-critical components with cyclic loading require fatigue analysis. Provides:
S-N life estimate, Miner's rule damage summation, fracture mechanics (LEFM) crack
growth, critical crack size, and inspection interval recommendation.

**Theoretical physics basis:**

Linear elastic fracture mechanics (LEFM). Stress intensity factor (SIF):
    KI = σ √(πa) · F(a/W)
where a = crack size, σ = remote stress, F = geometry correction factor (see handbooks).

Fracture occurs when: KI ≥ KIc (plane strain fracture toughness, material property)
Critical crack size: ac = (1/π)(KIc / (σ_max · F))²

Paris crack growth law (Paris & Erdogan 1963):
    da/dN = C · ΔK^m
where ΔK = Kmax - Kmin = Δσ √(πa) · F(a/W)
C, m are material constants (from fatigue crack growth test, ASTM E647)

Integration to fatigue crack life:
    Nf = ∫[a0 to ac] da / [C · ΔK(a)^m]

S-N (Wöhler) life for HCF (no mean stress correction):
    σa = σf' (2N)^b    (Basquin relation)
σf' = fatigue strength coefficient, b = fatigue strength exponent

Goodman mean stress correction:
    σa/σe + σm/σu = 1 → σe = σa / (1 - σm/σu)

## Step 1 — Load Spectrum and Stress History

```python
from dataclasses import dataclass
import numpy as np
from scipy import integrate

@dataclass
class LoadCycle:
    """Single load cycle definition. All stresses in MPa."""
    name: str
    sigma_max_MPa: float     # maximum stress (MPa)
    sigma_min_MPa: float     # minimum stress (MPa, can be negative for compression)
    n_cycles: float          # number of cycles in mission life
    frequency_Hz: float      # for loading rate effects

    @property
    def stress_range(self): return self.sigma_max_MPa - self.sigma_min_MPa
    @property
    def stress_ratio_R(self): return self.sigma_min_MPa / max(self.sigma_max_MPa, 1e-9)
    @property
    def mean_stress(self): return (self.sigma_max_MPa + self.sigma_min_MPa) / 2
    @property
    def stress_amplitude(self): return self.stress_range / 2
```

## Step 2 — S-N Fatigue Life (High Cycle)

```python
@dataclass
class FatigueMaterialData:
    """
    Fatigue material properties. MUST cite source — never recalled.
    Source: MMPDS, ASM Fatigue Database, manufacturer test data, or ASTM E466 result.
    """
    sigma_f_prime_MPa: float    # fatigue strength coefficient
    b_exponent: float           # fatigue strength exponent (typically -0.05 to -0.12)
    sigma_e_MPa: float          # endurance limit (at 10⁷ cycles; None for Al alloys)
    sigma_u_MPa: float          # ultimate tensile strength
    has_endurance_limit: bool   # True for steels <1400MPa, False for Al, Ti
    source: str                 # provenance

def sn_life_estimate(
    sigma_amplitude_MPa: float,
    sigma_mean_MPa: float,
    material: FatigueMaterialData,
    stress_concentration_Kt: float = 1.0,
    surface_finish_factor: float = 0.9,    # Machined=0.9, Ground=1.0, Rough=0.7
    size_factor: float = 0.85,             # for d > 8mm
) -> dict:
    """
    Basquin S-N life with Goodman mean stress correction.
    Applies Kt, surface finish, and size modification factors (Shigley's MDES method).

    Returns: Nf (cycles), infinite_life (bool), safety_factor_at_Nf
    """
    # Effective endurance limit with modification factors
    ka = surface_finish_factor
    kb = size_factor
    kc = 1.0   # loading factor (bending=1.0, torsion=0.577, axial=0.923)
    Se = material.sigma_e_MPa * ka * kb * kc / stress_concentration_Kt

    # Goodman correction: σ_eq = σ_a / (1 - σ_m/σ_u)
    if sigma_mean_MPa >= material.sigma_u_MPa:
        return {
            "Nf_cycles": 0,
            "status": "fail",
            "reason": f"Mean stress {sigma_mean_MPa:.1f} ≥ UTS {material.sigma_u_MPa:.1f} MPa — static fracture",
        }
    sigma_eq = sigma_amplitude_MPa / (1 - sigma_mean_MPa / material.sigma_u_MPa)

    if material.has_endurance_limit and sigma_eq <= Se:
        return {"Nf_cycles": float('inf'), "infinite_life": True, "Se_MPa": round(Se, 1)}

    # Basquin: σ_eq = σ_f' (2Nf)^b
    Nf = 0.5 * (sigma_eq / material.sigma_f_prime_MPa) ** (1 / material.b_exponent)

    return {
        "Nf_cycles": round(Nf, 0),
        "infinite_life": False,
        "sigma_eq_MPa": round(sigma_eq, 2),
        "Se_MPa": round(Se, 1),
        "status": "pass" if Nf > 1e7 else ("warn" if Nf > 1e5 else "fail"),
    }
```

## Step 3 — Miner's Rule Damage Accumulation

```python
def miners_rule(
    load_cycles: list[LoadCycle],
    material: FatigueMaterialData,
    stress_concentration_Kt: float = 1.0,
) -> dict:
    """
    Palmgren-Miner linear damage accumulation.
    D = Σ (ni/Ni)
    Failure when D = 1.0 (conservative; actual failure: D = 0.7–2.3)
    Use D_limit = 0.7 for safety-critical components.

    Reference: Miner (1945) J. Appl. Mech.
    """
    D_total = 0.0
    cycle_damage = []

    for lc in load_cycles:
        result = sn_life_estimate(
            lc.stress_amplitude, lc.mean_stress, material, stress_concentration_Kt
        )
        if result.get("infinite_life"):
            Ni = float('inf')
        else:
            Ni = result["Nf_cycles"]

        di = lc.n_cycles / Ni if Ni > 0 else float('inf')
        D_total += di
        cycle_damage.append({
            "name": lc.name, "ni": lc.n_cycles, "Ni": Ni,
            "di": round(di, 6), "contribution_pct": None,
        })

    for cd in cycle_damage:
        cd["contribution_pct"] = round(cd["di"] / max(D_total, 1e-9) * 100, 1)

    D_limit = 0.7   # safety-critical threshold
    status   = "pass" if D_total < D_limit else "fail"

    return {
        "damage_sum_D": round(D_total, 4),
        "D_limit": D_limit,
        "status": status,
        "life_fraction_used_pct": round(D_total / D_limit * 100, 1),
        "cycle_damage": cycle_damage,
    }
```

## Step 4 — Fracture Mechanics (LEFM, Paris Law)

```python
@dataclass
class FractureMaterialData:
    """
    LEFM material constants. MUST cite source (ASTM E647, MMPDS, literature).
    """
    KIc_MPa_m05: float      # plane-strain fracture toughness (MPa√m)
    C_paris: float          # Paris constant (m/cycle / (MPa√m)^m)
    m_paris: float          # Paris exponent (typically 2–4)
    source: str

def lefm_critical_crack_size(
    sigma_max_MPa: float,
    KIc_MPa_m05: float,
    geometry_factor_F: float = 1.0,    # F=1 for central through-crack, infinite plate
) -> float:
    """Critical crack half-length ac in meters."""
    return (1/np.pi) * (KIc_MPa_m05 / (sigma_max_MPa * geometry_factor_F))**2


def paris_crack_growth(
    a0_m: float,                    # initial crack size (NDE detection limit)
    ac_m: float,                    # critical crack size
    delta_sigma_MPa: float,         # stress range
    fracture_data: FractureMaterialData,
    geometry_factor_F: float = 1.0,
) -> dict:
    """
    Integrate Paris law from a0 to ac.
    Returns: Nf (cycles), inspection_interval_cycles

    Inspection interval = Nf / 2 (factor of 2 safety on crack growth life)
    Safe-life design: Nf / 4 (stricter, for critical structure)
    """
    C = fracture_data.C_paris
    m = fracture_data.m_paris

    def dN_da(a, _):
        """Inverse of da/dN for integration (da/dN = C·ΔK^m)."""
        delta_K = delta_sigma_MPa * np.sqrt(np.pi * a) * geometry_factor_F
        if delta_K <= 0:
            return 0
        da_dN = C * delta_K**m
        return 1 / da_dN   # dN/da

    result = integrate.solve_ivp(
        dN_da, [a0_m, ac_m], [0.0], max_step=ac_m/500, dense_output=True
    )
    Nf = float(result.y[0, -1])

    return {
        "Nf_crack_growth_cycles": round(Nf, 0),
        "a0_mm": a0_m * 1e3,
        "ac_mm": ac_m * 1e3,
        "inspection_interval_cycles": round(Nf / 2, 0),
        "safe_life_cycles": round(Nf / 4, 0),
        "status": "pass" if Nf > 1e6 else "fail",
    }
```

- Does **not** certify inspection intervals — NDI engineering authority sign-off required
