## Purpose

Chemical process engineering analysis for FORGE propulsion (bipropellant, monopropellant),
fuel systems, chemical manufacturing, and energy storage. Covers thermochemistry, reaction
kinetics, mass/energy balances, and process hazard analysis (HAZOP).

**Theoretical physics basis:**

First and second law of thermodynamics:
    dU = δQ - δW = TdS - pdV + Σμᵢdnᵢ
Gibbs free energy and equilibrium:
    dG = -SdT + Vdp + Σμᵢdnᵢ
Equilibrium: dG=0 → ΔGrxn° = -RT ln(K)
van't Hoff: d(lnK)/dT = ΔH°rxn/RT²

Reaction rate (Arrhenius):
    r = k(T) × f(Cᵢ),  k(T) = A·exp(-Ea/RT)
Activation energy Ea from ln(k) vs 1/T slope (Eyring theory for transition state)

Ideal rocket nozzle (Tsiolkovsky + isentropic):
    Isp = (1/g)·√(2γ/(γ-1) · R·Tc/M · [1-(pe/pc)^((γ-1)/γ)])

## Step 1 — Thermochemistry and Heat of Reaction

```python
from dataclasses import dataclass
import numpy as np

@dataclass
class ChemicalSpecies:
    """
    Thermodynamic data for a chemical species.
    Source: NIST-JANAF tables or CEA database. Never recalled.
    """
    name: str
    formula: str
    Hf0_kJ_mol: float          # standard enthalpy of formation (kJ/mol, 298K)
    Cp_J_mol_K: float          # heat capacity at constant pressure (avg over T range)
    MW_g_mol: float            # molecular weight
    source: str                # NIST-JANAF, CEA, DIPPR, etc.

def heat_of_reaction(
    reactants: list[tuple[float, ChemicalSpecies]],   # [(stoich_coeff, species)]
    products: list[tuple[float, ChemicalSpecies]],
    T_K: float = 298.15,
) -> dict:
    """
    Standard enthalpy of reaction (Hess's law) with Kirchhoff correction.
    ΔH°rxn = Σ(νᵢ Hf°products) - Σ(νᵢ Hf°reactants)
    Kirchhoff: ΔH(T) = ΔH°(298) + ∫[298 to T] ΔCp dT
    """
    dH_298 = (sum(nu * sp.Hf0_kJ_mol for nu, sp in products) -
              sum(nu * sp.Hf0_kJ_mol for nu, sp in reactants))

    # Kirchhoff correction (approximate, constant Cp)
    dCp = (sum(nu * sp.Cp_J_mol_K for nu, sp in products) -
           sum(nu * sp.Cp_J_mol_K for nu, sp in reactants))
    dH_T = dH_298 + dCp * (T_K - 298.15) / 1000   # kJ/mol

    reaction_type = "exothermic" if dH_T < 0 else "endothermic"

    return {
        "dH_298_kJ_mol": round(dH_298, 3),
        f"dH_{int(T_K)}K_kJ_mol": round(dH_T, 3),
        "type": reaction_type,
        "dCp_J_mol_K": round(dCp, 3),
    }
```

## Step 2 — Reaction Kinetics (Arrhenius)

```python
def arrhenius_rate(
    A_pre_exp: float,      # pre-exponential factor (units depend on reaction order)
    Ea_kJ_mol: float,      # activation energy
    T_K: float,
    concentrations: dict,  # {species: conc_mol_L}
    orders: dict,          # {species: reaction_order}
) -> dict:
    """
    Arrhenius rate law: r = A·exp(-Ea/RT) × Π[Cᵢ]^nᵢ

    R = 8.314 J/(mol·K)
    """
    R = 8.314e-3  # kJ/(mol·K)
    k = A_pre_exp * np.exp(-Ea_kJ_mol / (R * T_K))

    rate = k
    for species, order in orders.items():
        rate *= concentrations.get(species, 0)**order

    half_life = np.log(2) / k if len(orders) == 1 and list(orders.values())[0] == 1 else None

    return {
        "k": k,
        "rate_mol_L_s": rate,
        "half_life_s": round(half_life, 4) if half_life else "N/A (non-first-order)",
        "T_K": T_K,
    }
```

## Step 3 — Rocket Propellant Performance (Bipropellant)

```python
def bipropellant_isp(
    Tc_K: float,          # chamber temperature (from CEA or Nasa-Lewis code)
    gamma: float,         # specific heat ratio of combustion gases (CEA output)
    M_kg_mol: float,      # mean molecular weight of combustion gases (kg/mol)
    pc_Pa: float,         # chamber pressure
    pe_Pa: float,         # nozzle exit pressure (ambient or vacuum)
) -> dict:
    """
    Isentropic rocket nozzle performance.

    Specific impulse (s):
    Isp = (1/g0) × √(2γ/(γ-1) × R_u/M × Tc × [1-(pe/pc)^((γ-1)/γ)])

    Reference: Sutton & Biblarz "Rocket Propulsion Elements" 9th Ed.
    """
    g0  = 9.80665        # m/s²
    R_u = 8.314          # J/(mol·K)

    ve = np.sqrt(
        2 * gamma / (gamma - 1) * (R_u / M_kg_mol) * Tc_K *
        (1 - (pe_Pa/pc_Pa)**((gamma-1)/gamma))
    )
    Isp = ve / g0

    # Characteristic velocity c* (depends on Tc, not pe)
    c_star = np.sqrt(R_u * Tc_K / (M_kg_mol * gamma)) * (
        (2/(gamma+1))**((gamma+1)/(2*(gamma-1)))
    )**(-1)

    return {
        "Isp_s": round(Isp, 1),
        "ve_m_s": round(ve, 1),
        "c_star_m_s": round(c_star, 1),
        "Cf": round(ve / c_star, 4),   # thrust coefficient
        "source": "Sutton & Biblarz RPE 9th Ed., Eq. 3-30",
    }
```

## Step 4 — HAZOP Nodes

```python
def hazop_node_analysis(
    process_node: str,
    design_intent: str,
    parameters: list[str],   # e.g. ["flow", "temperature", "pressure"]
    guidewords: list[str] = None,
) -> list[dict]:
    """
    Structured HAZOP analysis using IEC 61882 guide words.
    Returns list of hazard scenarios with cause-consequence-safeguard.
    """
    if guidewords is None:
        guidewords = ["NO/NONE", "MORE", "LESS", "REVERSE", "PART OF",
                      "AS WELL AS", "OTHER THAN", "EARLY", "LATE"]

    scenarios = []
    for param in parameters:
        for gw in guidewords:
            deviation = f"{gw} {param}"
            scenarios.append({
                "node": process_node,
                "parameter": param,
                "guide_word": gw,
                "deviation": deviation,
                "cause": f"[ENGINEER: specify cause for '{deviation}' in node '{process_node}']",
                "consequence": f"[ENGINEER: specify consequence]",
                "safeguards": f"[ENGINEER: specify safeguards/instrumentation]",
                "action_required": f"[ENGINEER: add SIL rating if consequence = catastrophic]",
            })
    return scenarios
```
