## Purpose

Semiconductor device physics and power electronics for FORGE systems. Covers
pn junction analysis, MOSFET/IGBT device sizing for motor drives and power
conversion, wide-bandgap devices (GaN, SiC), and thermal management of power
semiconductors.

**Theoretical physics basis:**

Carrier statistics (Fermi-Dirac):
    f(E) = 1 / (1 + exp((E-EF)/kT))
Intrinsic carrier concentration:
    nᵢ = √(Nc·Nv) exp(-Eg/2kT)
    nᵢ(Si, 300K) = 9.65×10⁹ cm⁻³

PN junction built-in voltage:
    Vbi = (kT/q) ln(Na·Nd/nᵢ²)
Depletion width: W = √(2ε_s Vbi(Na+Nd)/(q·Na·Nd))

MOSFET threshold voltage:
    Vth = VFB + 2φF + Q_dep/Cox
    φF = (kT/q) ln(Na/nᵢ)

Bipolar transport (Shockley): J = J0(e^(qV/kT) - 1)
Avalanche breakdown: at E_field = Ecrit (SiC: 3MV/cm; Si: 0.3MV/cm; GaN: 3.3MV/cm)

## Step 1 — Power Device Selection

```python
from dataclasses import dataclass
import numpy as np

@dataclass
class PowerDeviceRequirements:
    """
    Power device requirements from system design.
    Source all electrical specifications from system-level design.
    """
    Vdc_bus_V: float         # DC bus voltage
    I_peak_A: float          # peak switch current
    I_rms_A: float           # RMS switch current
    f_sw_kHz: float          # switching frequency
    T_ambient_C: float       # ambient temperature
    T_junction_max_C: float  # max junction temperature (typ. 150–175°C for Si; 200°C for SiC)

def select_power_device(req: PowerDeviceRequirements) -> dict:
    """
    Device selection guide for power switches.
    Voltage rating = 1.5–2× DC bus voltage (safety margin for spikes).
    Current rating = 2× RMS current (derate for thermal margin).
    """
    V_rated = 1.7 * req.Vdc_bus_V   # 1.7× for spike margin
    I_rated = 2.0 * req.I_rms_A

    # Technology selection based on voltage/frequency
    if req.Vdc_bus_V < 100 and req.f_sw_kHz > 200:
        tech = "GaN_HEMT"
        rationale = "GaN: low Vth, near-zero reverse recovery, suitable for >200kHz"
        Rdson_estimate_mohm = 5 * (req.Vdc_bus_V / 100)**2.5  # empirical scaling
    elif req.Vdc_bus_V >= 600:
        tech = "SiC_MOSFET"
        rationale = "SiC: 3× higher Ecrit than Si, low switching losses at high V"
        Rdson_estimate_mohm = 10 * (req.Vdc_bus_V / 600)**2.3
    elif req.f_sw_kHz < 20 and req.I_rms_A > 100:
        tech = "Si_IGBT"
        rationale = "IGBT: high current density, low Vce(sat), but slow switching"
        Rdson_estimate_mohm = None
    else:
        tech = "Si_MOSFET"
        rationale = "Si MOSFET: well-characterised, cost-effective for <600V"
        Rdson_estimate_mohm = 20 * (req.Vdc_bus_V / 100)**2.7

    return {
        "recommended_technology": tech,
        "voltage_rating_V": round(V_rated, 0),
        "current_rating_A": round(I_rated, 1),
        "Rdson_estimate_mohm": round(Rdson_estimate_mohm, 1) if Rdson_estimate_mohm else "N/A (IGBT uses Vce_sat)",
        "rationale": rationale,
    }
```

## Step 2 — Conduction and Switching Losses

```python
def power_device_losses(
    I_rms_A: float,
    I_peak_A: float,
    Rdson_mohm: float,        # on-state resistance (MOSFET) OR Vce_sat_V for IGBT
    device_type: str,         # "MOSFET" or "IGBT"
    Vce_sat_V: float = 2.0,   # IGBT only
    Vdc_V: float = None,
    f_sw_Hz: float = None,
    t_rise_ns: float = None,  # switching rise time
    t_fall_ns: float = None,  # switching fall time
    duty_cycle: float = 0.5,
) -> dict:
    """
    Conduction and switching losses for power device.
    Conduction (MOSFET): P_cond = I_rms² × Rdson
    Conduction (IGBT):   P_cond = Vce_sat × I_rms × duty_cycle
    Switching:           P_sw = ½ × Vdc × I_peak × (t_r + t_f) × f_sw
    """
    if device_type == "MOSFET":
        P_cond = I_rms_A**2 * Rdson_mohm * 1e-3
    elif device_type == "IGBT":
        P_cond = Vce_sat_V * I_rms_A * duty_cycle
    else:
        P_cond = 0

    P_sw = 0
    if all(v is not None for v in [Vdc_V, f_sw_Hz, t_rise_ns, t_fall_ns]):
        P_sw = 0.5 * Vdc_V * I_peak_A * (t_rise_ns + t_fall_ns) * 1e-9 * f_sw_Hz

    return {
        "P_conduction_W": round(P_cond, 2),
        "P_switching_W": round(P_sw, 2),
        "P_total_W": round(P_cond + P_sw, 2),
        "efficiency_pct": round(100 * (1 - (P_cond + P_sw) / max(I_rms_A * Vdc_V * duty_cycle, 1e-3)), 2),
    }
```

## Step 3 — Thermal Resistance Stack

```python
def junction_temperature(
    P_total_W: float,
    Rth_jc_K_W: float,       # junction-to-case (from datasheet)
    Rth_cs_K_W: float = 0.1, # case-to-heatsink (thermal paste: ~0.05–0.2 K/W)
    Rth_sa_K_W: float = 1.0, # heatsink-to-ambient (heatsink spec or forced air)
    T_ambient_C: float = 40,
) -> dict:
    """
    Junction temperature estimate via thermal resistance chain.
    Tj = Ta + P × (Rth_jc + Rth_cs + Rth_sa)

    Derate if Tj > 0.75 × Tj_max (apply 25% derating for reliability)
    """
    R_total = Rth_jc_K_W + Rth_cs_K_W + Rth_sa_K_W
    Tj = T_ambient_C + P_total_W * R_total

    return {
        "Tj_C": round(Tj, 1),
        "R_total_K_W": round(R_total, 3),
        "dT_K": round(P_total_W * R_total, 1),
        "thermal_margin_C": round(150 - Tj, 1),  # assume Tj_max=150°C (Si), adjust for WBG
        "status": "pass" if Tj < 125 else ("warn" if Tj < 150 else "fail"),
    }
```
