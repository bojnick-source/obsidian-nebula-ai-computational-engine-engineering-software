## Purpose

Quantum engineering analysis for FORGE next-generation systems. Covers quantum
sensing (gravimeters, magnetometers, clocks), quantum communications, and quantum
computing hardware scoping. Not theoretical QM — applied engineering of quantum systems.

**Theoretical physics basis:**

Qubit state space: ℂ² (Hilbert space), |ψ⟩ = α|0⟩ + β|1⟩, |α|²+|β|²=1
Bloch sphere: |ψ⟩ = cos(θ/2)|0⟩ + e^(iφ)sin(θ/2)|1⟩

Decoherence:
    T₁ = energy relaxation time (qubit forgets its state)
    T₂ = dephasing time (T₂ ≤ 2T₁)
    T₂* = free-induction decay (inhomogeneous; T₂* < T₂)

Quantum Fisher Information (precision limit):
    ΔΦ ≥ 1/√(n F_Q)  (quantum Cramér-Rao bound)
    F_Q ≤ n (standard quantum limit); F_Q = n² (Heisenberg limit with entanglement)

Quantum sensing sensitivity (SQL):
    δB_min = ℏ/(γ √(T₂ N τ))  (magnetometer, γ=gyromagnetic ratio, N=n_atoms)

## Step 1 — Qubit Gate Fidelity Assessment

```python
import numpy as np

def qubit_gate_fidelity(
    T_gate_ns: float,        # gate time (ns)
    T1_us: float,            # T1 relaxation time (μs)
    T2_us: float,            # T2 dephasing time (μs)
    gate_error_ctrl: float,  # control error (calibration-limited, typ 1e-3)
) -> dict:
    """
    Gate fidelity estimate from decoherence and control errors.
    F ≈ 1 - p_T1 - p_T2 - p_ctrl

    Relaxation error per gate: p_T1 = T_gate / T1
    Dephasing error per gate:  p_T2 = T_gate / T2
    Total error: p_total = p_T1 + p_T2/2 + p_ctrl (approximate)

    Fault-tolerant threshold (surface code): error rate < ~1% per gate.
    """
    T_gate_us = T_gate_ns * 1e-3

    p_T1  = T_gate_us / T1_us
    p_T2  = T_gate_us / T2_us
    p_ctrl = gate_error_ctrl

    p_total = p_T1 + p_T2/2 + p_ctrl
    F_gate  = 1 - p_total

    return {
        "F_gate": round(F_gate, 6),
        "error_rate": round(p_total, 6),
        "p_T1": round(p_T1, 8),
        "p_T2": round(p_T2/2, 8),
        "p_ctrl": p_ctrl,
        "ft_threshold_met": p_total < 0.01,
        "note": "Surface code threshold ~1e-3; magic state distillation needs ~1e-4",
    }
```

## Step 2 — Quantum Sensor Performance

```python
def nv_center_magnetometer(
    N_centers: int,           # number of NV centers (spin ensemble)
    T2_star_us: float,        # T2* dephasing time (μs)
    tau_measurement_s: float, # total measurement time (s)
    eta_collection: float = 0.03,  # photon collection efficiency (typ 1–10%)
) -> dict:
    """
    NV-center ensemble DC magnetometer sensitivity.
    Sensitivity: δB = ℏ/(g_NV μ_B √(N T2* τ)) × 1/η

    NV properties:
    - Spin S=1, g_NV = 2.0028
    - Zero-field splitting D = 2.87 GHz
    - Hyperfine coupling A = 2.14 MHz (¹³C)
    - Photon detection contrast C ~ 0.03 (ODMR)

    Reference: Degen et al. Rev. Mod. Phys. 89 (2017)
    """
    hbar = 1.054571817e-34
    g_NV = 2.0028
    mu_B = 9.2740100783e-24  # J/T

    # Standard quantum limit for spin ensemble
    delta_B = hbar / (g_NV * mu_B * np.sqrt(N_centers * T2_star_us * 1e-6 * tau_measurement_s))
    delta_B_adjusted = delta_B / eta_collection  # photon shot noise

    return {
        "sensitivity_T_per_rootHz": delta_B_adjusted,
        "sensitivity_pT_per_rootHz": round(delta_B_adjusted * 1e12, 3),
        "T2_star_us": T2_star_us,
        "N_centers": N_centers,
        "state_of_art_2025": "~1 pT/√Hz (bulk diamond) → 100 fT/√Hz (atomic vapour cell)",
        "source": "Degen et al. Rev. Mod. Phys. 89, 035002 (2017)",
    }


def atomic_clock_stability(
    N_atoms: int,
    T_Ramsey_s: float,      # Ramsey interrogation time
    eta_detection: float,   # detection efficiency (0–1)
    f0_Hz: float,           # clock transition frequency
) -> dict:
    """
    Atomic clock Allan deviation at 1s (projection noise limit).
    σ_y(τ) = 1/(2πf₀) × 1/√(N T_R τ) × 1/η

    State of art 2025:
    - Optical lattice clocks (Sr, Yb): σ_y ~ 10⁻¹⁹ at 1000s
    - Chip-scale Cs clock (CSAC): σ_y ~ 10⁻¹⁰ at 1s
    """
    from scipy.constants import h, k
    sigma_y_1s = 1 / (2 * np.pi * f0_Hz * np.sqrt(N_atoms * T_Ramsey_s) * eta_detection)

    return {
        "Allan_deviation_1s": sigma_y_1s,
        "Allan_deviation_1s_log10": round(np.log10(sigma_y_1s), 2),
        "f0_Hz": f0_Hz,
        "N_atoms": N_atoms,
    }
```

## Step 3 — Quantum Key Distribution (BB84 Rate)

```python
def bb84_secure_key_rate(
    link_loss_dB: float,    # total channel + detection loss
    QBER: float,            # quantum bit error rate (0–0.5)
    f_Hz: float,            # pulse repetition rate (Hz)
    eta_detector: float = 0.15,  # detector efficiency (SNSPDs: 0.90; Si APD: 0.15)
) -> dict:
    """
    BB84 secure key rate (Scarani-Bechmann-Pasquinucci-Gisin formula, 2009).
    R_key = f × η_link × [1 - H(QBER) - H(QBER)]   (approximate)
    where H(e) = -e log₂(e) - (1-e) log₂(1-e) (binary entropy)
    Secure when QBER < 11% (Shor-Preskill criterion for BB84).

    Reference: Scarani et al. Rev. Mod. Phys. 81 (2009)
    """
    eta_link = 10**(-link_loss_dB/10) * eta_detector

    def H(e):
        if e <= 0 or e >= 1: return 0
        return -e * np.log2(e) - (1-e) * np.log2(1-e)

    # Sifted key rate (50% basis mismatch in BB84)
    R_sifted = 0.5 * f_Hz * eta_link

    # Privacy amplification + error correction
    R_secure = R_sifted * (1 - 2*H(QBER)) if QBER < 0.11 else 0

    return {
        "R_secure_bps": round(R_secure, 2),
        "R_sifted_bps": round(R_sifted, 2),
        "QBER_pct": round(QBER*100, 2),
        "secure": QBER < 0.11,
        "note": "SNSPD detectors (η=0.90) recommended for >100km links",
    }
```
