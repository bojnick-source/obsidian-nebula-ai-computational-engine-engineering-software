## Purpose

Uncertainty quantification (UQ), reliability analysis, Bayesian inference, and
statistical design for FORGE systems. Every engineering estimate has uncertainty —
this agent quantifies it. Used by all specialists to propagate uncertainty and
by verifier agents to assess confidence calibration.

**Core formalism:**

Probability axioms (Kolmogorov 1933):
    P(Ω) = 1, P(A) ≥ 0, P(A∪B) = P(A)+P(B) if A∩B = ∅

Bayes' theorem:
    P(θ|data) = P(data|θ) × P(θ) / P(data)
    posterior ∝ likelihood × prior

Central Limit Theorem:
    X̄_n → N(μ, σ²/n) as n→∞ (regardless of underlying distribution)

Failure probability (structural reliability):
    Pf = P(g(X) ≤ 0),  g(X) = R - S  (resistance - load)
Reliability index β: Pf = Φ(-β)

```python
import numpy as np
from scipy import stats

def monte_carlo_propagation(
    model: callable,           # scalar function of parameters
    param_distributions: list[dict],  # [{"name": str, "dist": stats.rv, "params": dict}]
    n_samples: int = 100_000,
    seed: int = 42,
) -> dict:
    """
    Monte Carlo uncertainty propagation.
    Draws n_samples from each input distribution, evaluates model,
    returns statistics of output distribution.

    When to use: nonlinear models, non-Gaussian inputs, large uncertainties
    Alternative: 1st-order (FOSM) when inputs are nearly Gaussian and model near-linear

    Example:
    param_distributions = [
        {"name": "E_GPa", "dist": stats.norm, "params": {"loc": 70, "scale": 3}},
        {"name": "t_mm",  "dist": stats.norm, "params": {"loc": 2.0, "scale": 0.1}},
    ]
    """
    rng = np.random.default_rng(seed)
    n_params = len(param_distributions)

    samples = np.zeros((n_samples, n_params))
    for j, pd in enumerate(param_distributions):
        samples[:, j] = pd["dist"].rvs(**pd["params"], size=n_samples, random_state=rng)

    outputs = np.array([model(*s) for s in samples])

    p05, p50, p95 = np.percentile(outputs, [5, 50, 95])

    return {
        "mean": round(float(outputs.mean()), 6),
        "std":  round(float(outputs.std()),  6),
        "CoV":  round(float(outputs.std() / abs(outputs.mean() + 1e-15)), 4),
        "p5":   round(float(p05), 6),
        "p50":  round(float(p50), 6),
        "p95":  round(float(p95), 6),
        "n_samples": n_samples,
        "ci_90_width": round(float(p95 - p05), 6),
    }


def fosm_propagation(
    model: callable,
    mean_params: dict,        # {"E": 70, "t": 2.0}
    std_params: dict,         # {"E": 3,  "t": 0.1}
) -> dict:
    """
    First-Order Second-Moment (FOSM) uncertainty propagation.
    Var(Y) ≈ Σᵢ (∂Y/∂xᵢ)² × σᵢ²  (linearised at mean)

    Valid for small CoV (σ/μ < 0.2) and smooth models.
    """
    names  = list(mean_params.keys())
    mu     = np.array([mean_params[n] for n in names])
    sigma  = np.array([std_params[n]  for n in names])

    Y_mean = model(*mu)

    # Numerical gradient
    eps = 1e-6
    gradients = []
    for i in range(len(names)):
        mu_p = mu.copy(); mu_p[i] += eps
        mu_m = mu.copy(); mu_m[i] -= eps
        dY_dxi = (model(*mu_p) - model(*mu_m)) / (2*eps)
        gradients.append(dY_dxi)

    variance = sum(g**2 * s**2 for g, s in zip(gradients, sigma))

    return {
        "Y_mean": round(Y_mean, 6),
        "Y_std": round(np.sqrt(variance), 6),
        "CoV": round(np.sqrt(variance) / abs(Y_mean + 1e-15), 4),
        "sensitivity_indices": {n: round(abs(g*s)/np.sqrt(variance + 1e-15), 4)
                                 for n, g, s in zip(names, gradients, sigma)},
    }
```

```python
def hasofer_lind_reliability(
    g_func: callable,       # limit state g(X) < 0 = failure
    means: np.ndarray,
    cov_matrix: np.ndarray,
    max_iter: int = 100,
) -> dict:
    """
    Hasofer-Lind FORM (First Order Reliability Method).
    β = min ||u||₂  subject to g(T(u)) = 0
    Pf = Φ(-β)

    Transforms to standard normal space: u = L⁻¹(X - μ), L from Cholesky
    Uses HLRF algorithm for constrained minimisation.

    Reference: Hasofer & Lind (1974) J. Eng. Mech. Div.
    """
    L = np.linalg.cholesky(cov_matrix)
    L_inv = np.linalg.inv(L)

    u = np.zeros(len(means))  # start at mean in standard space

    for iteration in range(max_iter):
        X = means + L @ u

        g_val = g_func(X)
        if abs(g_val) < 1e-8:
            break

        # Gradient in u-space
        eps = 1e-6
        n = len(u)
        grad_g_x = np.zeros(n)
        for i in range(n):
            X_p = X.copy(); X_p[i] += eps
            X_m = X.copy(); X_m[i] -= eps
            grad_g_x[i] = (g_func(X_p) - g_func(X_m)) / (2*eps)

        grad_g_u = L.T @ grad_g_x

        # HLRF step
        alpha   = -grad_g_u / (np.linalg.norm(grad_g_u) + 1e-15)
        beta_k  = np.linalg.norm(u)
        u_new   = (np.dot(grad_g_u, u) - g_val) / np.dot(grad_g_u, grad_g_u) * grad_g_u

        if np.linalg.norm(u_new - u) < 1e-8:
            u = u_new
            break
        u = u_new

    beta = np.linalg.norm(u)
    from scipy.stats import norm as scipy_norm
    Pf   = scipy_norm.cdf(-beta)

    return {
        "beta": round(beta, 4),
        "Pf": Pf,
        "Pf_log10": round(np.log10(max(Pf, 1e-15)), 2),
        "design_point_X": X,
        "alpha_direction_cosines": -u / (np.linalg.norm(u) + 1e-15),
        "converged": abs(g_func(X)) < 1e-6,
    }


def weibull_fit_and_life(
    failure_times: np.ndarray,   # observed failure times (hours or cycles)
    confidence: float = 0.90,
) -> dict:
    """
    Fit 2-parameter Weibull distribution to failure data.
    F(t) = 1 - exp(-(t/η)^β)  where η=scale, β=shape

    β < 1: early failures (infant mortality)
    β = 1: constant rate (Poisson, memoryless)
    β > 1: wear-out failures (fatigue, ageing)

    Returns: B10 life (10% failure), B50, reliability at given time
    """
    shape, loc, scale = stats.weibull_min.fit(failure_times, floc=0)
    # scipy shape=β, scale=η

    B10 = scale * (-np.log(0.9))**(1/shape)
    B50 = scale * (-np.log(0.5))**(1/shape)

    return {
        "beta_shape": round(shape, 4),
        "eta_scale": round(scale, 4),
        "B10_life": round(B10, 2),
        "B50_life": round(B50, 2),
        "failure_mode": ("infant_mortality" if shape < 1 else
                         "random" if abs(shape-1) < 0.2 else "wearout"),
        "source": "scipy weibull_min MLE fit",
    }
```

## Step 3 — Bayesian Inference (MCMC)

```python
def metropolis_hastings(
    log_posterior: callable,   # log P(θ|data) = log likelihood + log prior
    theta_init: np.ndarray,
    proposal_std: np.ndarray,  # step size per parameter
    n_samples: int = 50_000,
    burn_in: int = 10_000,
) -> dict:
    """
    Metropolis-Hastings MCMC sampler.
    Samples from posterior P(θ|data) without normalising constant.

    Acceptance rate target: 0.234 for optimal mixing (Roberts et al. 1997).
    If acceptance < 0.1: proposal_std too large.
    If acceptance > 0.5: proposal_std too small (slow mixing).
    """
    n_params = len(theta_init)
    samples  = np.zeros((n_samples, n_params))
    theta    = theta_init.copy()
    lp       = log_posterior(theta)
    accepted = 0

    for i in range(n_samples + burn_in):
        theta_prop = theta + np.random.randn(n_params) * proposal_std
        lp_prop    = log_posterior(theta_prop)

        if np.log(np.random.rand()) < lp_prop - lp:
            theta = theta_prop
            lp    = lp_prop
            accepted += 1

        if i >= burn_in:
            samples[i - burn_in] = theta

    acceptance_rate = accepted / (n_samples + burn_in)

    return {
        "samples": samples,
        "means": samples.mean(axis=0),
        "stds":  samples.std(axis=0),
        "acceptance_rate": round(acceptance_rate, 4),
        "warning": "Adjust proposal_std" if acceptance_rate < 0.1 or acceptance_rate > 0.5 else None,
    }
```

## Sobol Sensitivity Indices (Variance-Based)

For model Y = f(X₁, ..., Xₖ):
- First-order Sᵢ = Var(E[Y|Xᵢ]) / Var(Y) — main effect
- Total effect STᵢ = 1 - Var(E[Y|X_{-i}]) / Var(Y) — including interactions
- Interaction: STᵢ - Sᵢ > 0 indicates parameter interactions

Use saltelli.sample from SALib library for efficient Sobol sampling.
Flag `[UQ: HIGH INTERACTION]` if max(STᵢ - Sᵢ) > 0.1.
