## Purpose

Environmental impact assessment for FORGE designs. Covers life cycle assessment (LCA),
GHG emissions, material toxicity/regulatory compliance (REACH, RoHS), noise/pollution,
and sustainability metrics. Required for products entering EU, US, or global markets.

**Theoretical physics basis:**

Life cycle energy flows (first law):
    E_lifecycle = E_material + E_manufacturing + E_use + E_disposal
Greenhouse gas characterisation (GWP100):
    GWP100_CO2eq = Σᵢ mᵢ × GWPᵢ
    CO2=1, CH4≈28, N2O≈265, HFCs=hundreds to thousands (IPCC AR6)

Thermodynamic exergy (second law):
    Ex = (H-H0) - T0(S-S0)  (maximum useful work)
Cumulative Exergy Demand (CExD): measures resource scarcity, not just energy

## Step 1 — Life Cycle Assessment (ISO 14040/44 Framework)

```python
from dataclasses import dataclass, field
import numpy as np
from typing import Optional

@dataclass
class LCAPhase:
    """Single LCA phase data. Source all from EcoInvent 3.10 or EPD."""
    phase: str                    # material | manufacturing | use | EoL
    CO2eq_kg: float               # kg CO2-equivalent (GWP100, IPCC AR6)
    energy_MJ: float              # primary energy demand (MJ)
    water_L: float = 0.0          # water consumption (L)
    source: str = ""              # EcoInvent UUID or EPD reference

@dataclass
class LifeCycleInventory:
    """Complete LCI for a component."""
    component_id: str
    material_kg: float
    phases: list[LCAPhase] = field(default_factory=list)

def lca_summary(lci: LifeCycleInventory) -> dict:
    """Aggregate LCA results across all phases."""
    total_CO2_kg  = sum(p.CO2eq_kg for p in lci.phases)
    total_E_MJ    = sum(p.energy_MJ for p in lci.phases)
    total_water_L = sum(p.water_L for p in lci.phases)

    co2_by_phase = {p.phase: round(p.CO2eq_kg, 3) for p in lci.phases}
    e_by_phase   = {p.phase: round(p.energy_MJ, 2) for p in lci.phases}

    # Hotspot identification
    dominant = max(co2_by_phase, key=co2_by_phase.get)

    return {
        "total_CO2eq_kg": round(total_CO2_kg, 2),
        "total_primary_energy_MJ": round(total_E_MJ, 2),
        "total_water_L": round(total_water_L, 1),
        "CO2_by_phase": co2_by_phase,
        "energy_by_phase": e_by_phase,
        "hotspot_phase": dominant,
        "CO2_per_kg_component": round(total_CO2_kg / max(lci.material_kg, 0.001), 2),
        "note": "Cross-check with EPD from manufacturer; EcoInvent background data only",
    }
```

## Step 2 — GHG Emissions Inventory (Scope 1/2/3)

```python
def scope_emissions(
    electricity_kWh: float,
    grid_emission_factor_kg_kWh: float = 0.4,   # kg CO2/kWh (UK 2025 avg: ~0.22)
    fuel_L: float = 0.0,
    fuel_type: str = "diesel",
    flights_kgCO2: float = 0.0,
    supply_chain_spend_kUSD: float = 0.0,
    emission_intensity_kg_kUSD: float = 200.0,  # EEIO estimate
) -> dict:
    """
    GHG Protocol Scope 1/2/3 emissions estimate.
    Scope 1: direct combustion (fuel_L)
    Scope 2: grid electricity
    Scope 3: supply chain, travel
    """
    fuel_EF = {"diesel": 2.68, "petrol": 2.31, "LNG": 2.75,
               "kerosene": 2.54, "LPG": 1.51}   # kg CO2/L (DEFRA 2024)

    scope1 = fuel_L * fuel_EF.get(fuel_type, 2.5)
    scope2 = electricity_kWh * grid_emission_factor_kg_kWh
    scope3 = flights_kgCO2 + supply_chain_spend_kUSD * emission_intensity_kg_kUSD

    total = scope1 + scope2 + scope3

    return {
        "scope1_kg_CO2": round(scope1, 2),
        "scope2_kg_CO2": round(scope2, 2),
        "scope3_kg_CO2": round(scope3, 2),
        "total_kg_CO2eq": round(total, 2),
        "sources": "DEFRA 2024 conversion factors; GHG Protocol 2015",
    }
```

## Step 3 — REACH and RoHS Compliance Check

```python
def reach_rohs_check(materials_used: list[dict]) -> dict:
    """
    Flag Substances of Very High Concern (SVHC) per REACH XXIX candidate list (2024)
    and RoHS III (Directive 2015/863) restricted substances.

    materials_used: [{"name": str, "CAS": str, "mass_kg": float}]

    NOTE: SVHC list updated twice yearly. Verify against ECHA candidate list.
    """
    # SVHC substances (subset — check ECHA for complete current list)
    SVHC_CAS = {
        "1763-23-1": "PFOS (perfluorooctane sulfonate)",
        "375-73-5":  "PFOS-related",
        "25637-99-4": "Hexabromocyclododecane (HBCD)",
        "7789-06-2": "Strontium chromate",
        "10588-01-9": "Sodium dichromate",
        "7778-50-9": "Potassium dichromate",
        "1333-82-0": "Chromium trioxide",
    }
    # RoHS restricted (Annex II, Directive 2015/863)
    RoHS_CAS = {
        "7439-92-1": "Lead",
        "7439-97-6": "Mercury",
        "7440-43-9": "Cadmium",
        "7440-02-0": "Nickel",    # not restricted but notable
        "18540-29-9": "Chromium(VI)",
    }
    RoHS_limits = {"7439-92-1": 0.1, "7439-97-6": 0.1, "7440-43-9": 0.01,
                   "18540-29-9": 0.1}  # wt% in homogeneous material

    flags = []
    for mat in materials_used:
        cas = mat.get("CAS", "")
        if cas in SVHC_CAS:
            flags.append({
                "type": "SVHC",
                "material": mat["name"],
                "CAS": cas,
                "substance": SVHC_CAS[cas],
                "action": "Notify supply chain >0.1wt%; substitute or apply ECHA authorisation",
            })
        if cas in RoHS_limits:
            flags.append({
                "type": "RoHS_restricted",
                "material": mat["name"],
                "CAS": cas,
                "limit_wt_pct": RoHS_limits[cas],
                "action": "Verify concentration <limit in homogeneous material",
            })

    return {
        "flags": flags,
        "n_SVHC_flags": sum(1 for f in flags if f["type"] == "SVHC"),
        "n_RoHS_flags": sum(1 for f in flags if f["type"] == "RoHS_restricted"),
        "status": "clean" if not flags else "action_required",
        "reference": "ECHA SVHC Candidate List (Jan 2025); EU RoHS III 2015/863",
    }
```
