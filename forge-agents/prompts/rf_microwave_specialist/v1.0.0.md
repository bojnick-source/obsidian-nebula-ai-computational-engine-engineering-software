## Purpose

RF/microwave engineering analysis for FORGE systems with wireless communication,
radar, telemetry, or EMC requirements. Provides link budget, antenna gain, noise
figure, transmission line design, and EMC compliance.

**Theoretical physics basis — Maxwell's equations:**
    ∇×H = J + ∂D/∂t    (Ampère-Maxwell)
    ∇×E = -∂B/∂t       (Faraday)
    ∇·D = ρ             (Gauss electric)
    ∇·B = 0             (Gauss magnetic)

Wave equation in source-free region:
    ∇²E - με ∂²E/∂t² = 0 → E(z,t) = E₀ e^{j(ωt-βz)}
Propagation constant: β = ω√(με) = 2π/λ

Friis transmission equation:
    Pr/Pt = Gt · Gr · (λ/4πd)²
Free-space path loss: FSPL(dB) = 20log₁₀(4πd/λ)

Nyquist noise power: N = kTB (k=Boltzmann, T=temp K, B=bandwidth Hz)

## Step 1 — Link Budget Analysis

```python
import numpy as np
from dataclasses import dataclass

@dataclass
class LinkBudget:
    """
    Complete RF link budget. All power in dBm, gains/losses in dB.
    Cite source for each parameter (datasheet, ITU-R, measured).
    """
    f_GHz: float              # carrier frequency
    d_km: float               # link distance
    Pt_dBm: float             # transmit power
    Gt_dBi: float             # transmit antenna gain
    Gr_dBi: float             # receive antenna gain
    Lt_dB: float              # transmit feed losses
    Lr_dB: float              # receive feed losses
    T_sys_K: float            # system noise temperature
    B_MHz: float              # noise bandwidth
    SNR_req_dB: float         # required SNR for target BER/modulation
    margin_req_dB: float = 6  # fade margin (link reliability target)

def compute_link_budget(lb: LinkBudget) -> dict:
    """Friis link budget. Returns margin and status."""
    lam_m = 0.3 / lb.f_GHz   # wavelength (m)
    d_m   = lb.d_km * 1e3

    FSPL_dB = 20 * np.log10(4 * np.pi * d_m / lam_m)

    # Received signal power
    Pr_dBm = lb.Pt_dBm + lb.Gt_dBi - lb.Lt_dB - FSPL_dB + lb.Gr_dBi - lb.Lr_dB

    # Noise floor
    k_dBm = -174.0   # kTB at 290K, 1Hz bandwidth in dBm/Hz
    N_dBm = k_dBm + 10*np.log10(lb.T_sys_K/290) + 10*np.log10(lb.B_MHz * 1e6)

    SNR_achieved_dB = Pr_dBm - N_dBm
    margin_dB = SNR_achieved_dB - lb.SNR_req_dB

    return {
        "FSPL_dB": round(FSPL_dB, 1),
        "Pr_dBm": round(Pr_dBm, 2),
        "N_dBm": round(N_dBm, 2),
        "SNR_achieved_dB": round(SNR_achieved_dB, 2),
        "SNR_required_dB": lb.SNR_req_dB,
        "link_margin_dB": round(margin_dB, 2),
        "status": "pass" if margin_dB >= lb.margin_req_dB else "fail",
        "wavelength_mm": round(lam_m * 1000, 2),
    }
```

## Step 2 — Antenna Design

```python
def dipole_gain_dBi(L_lambda: float = 0.5) -> float:
    """Half-wave dipole gain: 2.15 dBi. Quarter-wave: ~2 dBi."""
    if abs(L_lambda - 0.5) < 0.05:
        return 2.15
    elif abs(L_lambda - 0.25) < 0.03:
        return 2.0
    else:
        return 2.15  # approximate

def patch_antenna_design(
    f_GHz: float,
    er: float = 4.4,    # substrate permittivity (FR4=4.4, Rogers 4003=3.55)
    h_mm: float = 1.6,  # substrate thickness
) -> dict:
    """
    Rectangular microstrip patch antenna design.
    Reference: Balanis "Antenna Theory" 3rd Ed., Chapter 14.

    Returns: patch width W, length L, feed position for 50Ω input.
    """
    lam0_mm = 300 / f_GHz       # free-space wavelength (mm)
    c0_mm_s = 3e11              # mm/s

    # Effective permittivity (Hammerstad-Jensen)
    W = (c0_mm_s / (2 * f_GHz * 1e9)) * np.sqrt(2 / (er + 1))   # mm
    er_eff = (er+1)/2 + (er-1)/2 * (1 + 12*h_mm/W)**(-0.5)

    # Effective length extension (Hammerstad formula)
    delta_L = 0.412 * h_mm * (er_eff+0.3)*(W/h_mm+0.264) / ((er_eff-0.258)*(W/h_mm+0.8))

    # Resonant length
    L = c0_mm_s / (2 * f_GHz * 1e9 * np.sqrt(er_eff)) - 2*delta_L

    # Input impedance at edge ≈ Rin (varies 100–400Ω)
    # Feed offset for 50Ω: x0 = L/π * arccos(√(50/Rin))
    Rin_approx = 90 * er**2 / (er-1) * (L/W)**2   # Munson approximation
    import numpy as np
    x0 = L / np.pi * np.arccos(np.sqrt(50 / Rin_approx)) if Rin_approx > 50 else L/2

    return {
        "W_mm": round(W, 2),
        "L_mm": round(L, 2),
        "h_mm": h_mm,
        "er": er,
        "er_eff": round(er_eff, 3),
        "feed_offset_mm": round(x0, 2),
        "gain_estimate_dBi": 6.0 + 10*np.log10(er_eff) * 0.3,  # ~6-8 dBi typical
    }
```

## Step 3 — Transmission Line and Impedance Matching

```python
def microstrip_Z(
    W_mm: float, h_mm: float, er: float,
) -> float:
    """Microstrip characteristic impedance Z0 (Wheeler 1977 formula)."""
    u = W_mm / h_mm
    if u <= 1:
        F = 6 + (2*np.pi-6)*np.exp(-(30.666/u)**0.7528)
        Z0 = 60 * np.log(F/u + np.sqrt(1 + (2/u)**2)) / np.sqrt((er+1)/2 + (er-1)/2*(1+10/u)**(-0.5))
    else:
        er_eff = (er+1)/2 + (er-1)/2*(1+10/u)**(-0.5)
        Z0 = 120*np.pi / (np.sqrt(er_eff)*(u + 1.393 + 0.667*np.log(u+1.444)))
    return Z0

def vswr_from_gamma(gamma_mag: float) -> float:
    """VSWR = (1+|Γ|)/(1-|Γ|). Γ = (ZL-Z0)/(ZL+Z0)."""
    return (1 + gamma_mag) / (1 - gamma_mag)

def return_loss_dB(gamma_mag: float) -> float:
    """Return loss RL = -20log₁₀|Γ| (positive dB = good)."""
    return -20 * np.log10(max(gamma_mag, 1e-9))
```

## EMC Assessment

```python
def emi_radiated_limit_check(
    E_field_dBuV_m: float,    # measured/simulated E-field at 3m (or specified distance)
    frequency_MHz: float,
    standard: str = "FCC_Part15_B",   # FCC_Part15_B | CISPR_32_ClassB | MIL_STD_461G_RE102
) -> dict:
    """
    Check radiated emissions against standard limit.
    Limits are approximate and frequency-dependent.
    """
    limits = {
        "FCC_Part15_B": {  # at 3m, Class B (residential)
            (30, 88):   40.0,
            (88, 216):  43.5,
            (216, 960): 46.0,
            (960, 1e6): 54.0,
        },
        "MIL_STD_461G_RE102": {  # 1m, narrowband limits (platform-dependent)
            (2, 30):    24.0,
            (30, 1000): 34.0,
            (1000, 18000): 40.0,
        },
    }
    limit = None
    for (f_lo, f_hi), lim_val in limits.get(standard, {}).items():
        if f_lo <= frequency_MHz < f_hi:
            limit = lim_val
            break

    if limit is None:
        return {"status": "unknown", "note": f"No limit found for {standard} at {frequency_MHz}MHz"}

    margin = limit - E_field_dBuV_m
    return {
        "limit_dBuV_m": limit,
        "measured_dBuV_m": E_field_dBuV_m,
        "margin_dB": round(margin, 1),
        "status": "pass" if margin >= 0 else "fail",
        "standard": standard,
    }
```
