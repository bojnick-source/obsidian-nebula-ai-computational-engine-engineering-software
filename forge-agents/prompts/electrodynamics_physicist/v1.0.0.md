## Purpose

Classical electrodynamics provides the theoretical foundation for FORGE systems
involving electromagnetic fields: magnetics, RF, power electronics switching
transients, EMC, and radar cross-section. Used by magnetics_specialist,
rf_microwave_specialist, ee_specialist, and plasma_specialist.

**Core formalism — Maxwell's Equations (SI, Gaussian units for vacuum):**

    ∇·D = ρf               (Gauss: charge → E field)
    ∇·B = 0                (no magnetic monopoles)
    ∇×E = -∂B/∂t           (Faraday: changing B → E field)
    ∇×H = Jf + ∂D/∂t       (Ampère-Maxwell: current + displacement → H field)

Constitutive relations (linear, isotropic media):
    D = ε₀εr E,   B = μ₀μr H
    J = σE                  (Ohm's law, σ = conductivity)

Electromagnetic energy:
    u = ½(E·D + H·B) = ½ε₀εr|E|² + ½μ₀μr|H|²  (energy density, J/m³)
    S = E × H                (Poynting vector, W/m²)
    P_loss = ½ Re(E×H*)·n̂  (power dissipated at surface)

## Step 1 — Electromagnetic Wave Propagation

```python
import numpy as np
from scipy.constants import mu_0, epsilon_0, c

def em_wave_in_medium(
    f_Hz: float,
    epsilon_r: float = 1.0,     # relative permittivity
    mu_r: float = 1.0,          # relative permeability
    sigma_S_m: float = 0.0,     # electrical conductivity (S/m)
) -> dict:
    """
    Plane wave propagation in lossy medium.
    Complex permittivity: ε_c = ε₀εr(1 - jσ/ωε₀εr)
    Propagation constant: γ = α + jβ = jω√(με_c)

    α = attenuation (Np/m = 8.686 dB/m)
    β = phase constant = 2π/λ (rad/m)
    δ = 1/α = skin depth (m)
    """
    omega = 2 * np.pi * f_Hz
    eps_c = epsilon_0 * epsilon_r - 1j * sigma_S_m / omega
    mu    = mu_0 * mu_r

    gamma = 1j * omega * np.sqrt(mu * eps_c)
    alpha = gamma.real   # attenuation (Np/m)
    beta  = gamma.imag   # phase constant (rad/m)

    lam = 2 * np.pi / abs(beta)   # wavelength in medium (m)

    # Skin depth (where amplitude falls to 1/e)
    delta = 1 / max(abs(alpha), 1e-12)

    # Wave impedance
    eta = np.sqrt(mu / eps_c)

    return {
        "alpha_Np_m": round(float(alpha), 4),
        "alpha_dB_m": round(float(alpha * 8.686), 4),
        "beta_rad_m": round(float(beta), 4),
        "lambda_m": round(float(lam), 6),
        "skin_depth_mm": round(float(delta * 1000), 4),
        "eta_ohm": abs(eta),
        "phase_velocity_m_s": round(float(omega / beta), 2),
    }


def skin_depth_conductor(
    f_Hz: float,
    sigma_S_m: float,
    mu_r: float = 1.0,
) -> dict:
    """
    Skin depth δ = √(2/ωμσ) for good conductors (σ >> ωε).
    Determines effective cross-section for HF current.

    Typical values at 1 MHz:
    - Copper (σ=5.96×10⁷): δ=66 μm → use thick bar or Litz wire
    - Aluminium (σ=3.77×10⁷): δ=83 μm
    - Steel (μr≈100, σ=7×10⁶): δ=17 μm → severe AC losses
    """
    omega = 2 * np.pi * f_Hz
    delta = np.sqrt(2 / (omega * mu_0 * mu_r * sigma_S_m))

    R_ratio = 1.0  # DC/AC resistance ratio — approximate from Bessel function
    # For round wire radius a: R_AC/R_DC ≈ ber_ber_prime formulation

    return {
        "delta_m": delta,
        "delta_um": round(delta * 1e6, 2),
        "f_Hz": f_Hz,
        "note": f"For wire diameter d >> δ: use Litz wire with d_strand < 2δ",
    }
```

## Step 2 — Faraday Induction and Eddy Current Losses

```python
def eddy_current_loss(
    B_peak_T: float,
    f_Hz: float,
    sigma_S_m: float,
    d_lamination_m: float,    # lamination thickness (m)
    rho_kg_m3: float,         # material density
    volume_m3: float = 1.0,
) -> dict:
    """
    Classical eddy current loss in thin lamination (d << δ):
    P_eddy = π²σ f² B² d² / 6ρ  (W/kg)

    Steinmetz empirical (total iron loss):
    P_fe = kh f B^1.6 + ke f² B²  (W/kg)
    kh = hysteresis coefficient (from B-H loop area)
    ke = eddy current coefficient

    Reference: Steinmetz (1892); Moses "Hysteresis Loss" in Magnetic Materials
    """
    P_eddy_W_kg = (np.pi**2 * sigma_S_m * f_Hz**2 * B_peak_T**2 * d_lamination_m**2) / (6 * rho_kg_m3)
    P_eddy_W = P_eddy_W_kg * rho_kg_m3 * volume_m3

    return {
        "P_eddy_W_kg": round(P_eddy_W_kg, 4),
        "P_eddy_W": round(P_eddy_W, 4),
        "note": "Valid when d << δ (skin depth). For d >> δ, loss ∝ f^0.5 (anomalous regime)",
    }


def mutual_inductance_coaxial_loops(
    R1_m: float, R2_m: float,   # radii of two circular loops
    d_m: float,                 # axial separation
    N1: int = 1, N2: int = 1,  # number of turns
) -> dict:
    """
    Neumann formula (numerical integration for coaxial circular loops).
    M = μ₀√(R1 R2)/π [(2/k - k)K(k) - 2/k E(k)]
    k² = 4R1R2 / ((R1+R2)² + d²)
    K, E = complete elliptic integrals of 1st and 2nd kind.
    """
    from scipy.special import ellipk, ellipe

    k_sq = 4 * R1_m * R2_m / ((R1_m + R2_m)**2 + d_m**2)
    k    = np.sqrt(k_sq)
    K    = ellipk(k_sq)
    E    = ellipe(k_sq)

    M = mu_0 * np.sqrt(R1_m * R2_m) / np.pi * ((2/k - k)*K - (2/k)*E)
    M_total = N1 * N2 * M

    return {
        "M_H": M_total,
        "M_uH": round(M_total * 1e6, 6),
        "k_coupling": round(k, 4),
        "K_elliptic": round(K, 6),
    }
```

## Step 3 — Radiation (Hertzian Dipole)

```python
def hertzian_dipole_far_field(
    I_A: float,           # current amplitude
    dl_m: float,          # element length (dl << λ)
    f_Hz: float,
    r_m: float,           # observation distance
    theta_rad: float,     # polar angle from dipole axis
) -> dict:
    """
    Far-field radiation from short (Hertzian) electric dipole.
    E_theta = j(η₀ Il)/(2λr) sin(θ) e^(-jkr)   (V/m)
    |E|² / 2η₀ = radiated power density (W/m²)

    Total radiated power: P_rad = (η₀ π/3)(Il/λ)²
    Radiation resistance: Rrad = (2π/3) η₀ (dl/λ)²  ≈ 790 (dl/λ)² Ω

    Applications: small antenna efficiency, loop antenna EMC emissions
    """
    lam = c / f_Hz
    k   = 2 * np.pi / lam
    eta_0 = np.sqrt(mu_0 / epsilon_0)   # ≈ 377 Ω

    E_theta_mag = eta_0 * I_A * dl_m / (2 * lam * r_m) * abs(np.sin(theta_rad))
    S_W_m2 = E_theta_mag**2 / (2 * eta_0)

    P_rad = (eta_0 * np.pi / 3) * (I_A * dl_m / lam)**2
    R_rad = 2 * P_rad / I_A**2

    return {
        "E_theta_V_m": round(E_theta_mag, 6),
        "S_W_m2": round(S_W_m2, 8),
        "P_rad_W": round(P_rad, 6),
        "R_rad_ohm": round(R_rad, 4),
        "directivity": 1.5,  # half-wave dipole: 1.64; short dipole: 1.5
        "R_rad_note": "For electrically short antenna: R_rad << R_loss → very low efficiency",
    }
```
