## Purpose

Designs and analyses fibre-reinforced polymer (FRP) composite laminates.
Covers ply-by-ply micromechanics, classical laminate theory (CLT), failure
prediction, and manufacturing constraints. Provides ABD matrix, laminate
stiffness, and Tsai-Wu failure indices for each ply.

**Theoretical physics basis:**

Constitutive relations in principal material axes (1-2-3):
    σ₁ = Q₁₁ε₁ + Q₁₂ε₂
    σ₂ = Q₁₂ε₁ + Q₂₂ε₂
    τ₁₂ = Q₆₆γ₁₂

where the reduced stiffness matrix Q:
    Q₁₁ = E₁/(1-ν₁₂ν₂₁),  Q₂₂ = E₂/(1-ν₁₂ν₂₁),
    Q₁₂ = ν₁₂E₂/(1-ν₁₂ν₂₁),  Q₆₆ = G₁₂

Transformation to global axes (θ = ply angle):
    Q̄ᵢⱼ = T⁻¹ Qᵢⱼ T⁻ᵀ
where T is the direction cosine transformation matrix.

Laminate stiffnesses (CLT):
    Aᵢⱼ = Σₖ Q̄ᵢⱼ^(k) hₖ            (extensional stiffness, N/m)
    Bᵢⱼ = ½ Σₖ Q̄ᵢⱼ^(k) (zₖ² - zₖ₋₁²)  (bending-extension coupling)
    Dᵢⱼ = ⅓ Σₖ Q̄ᵢⱼ^(k) (zₖ³ - zₖ₋₁³)  (bending stiffness, N·m)

## Step 1 — Micromechanics (Ply Properties from Constituents)

```python
from dataclasses import dataclass
import numpy as np

@dataclass
class FibreProperties:
    """Fibre material properties. Source: manufacturer datasheet required."""
    E_f_GPa: float        # Axial modulus
    E_ft_GPa: float       # Transverse modulus
    nu_f: float           # Poisson's ratio
    G_f_GPa: float        # Shear modulus
    rho_kg_m3: float      # Density

@dataclass
class MatrixProperties:
    """Matrix (resin) properties. Source: datasheet required."""
    E_m_GPa: float
    nu_m: float
    G_m_GPa: float
    rho_kg_m3: float
    T_g_C: float          # Glass transition temperature (critical: use T < T_g - 30°C)

def micromechanics(
    fibre: FibreProperties,
    matrix: MatrixProperties,
    Vf: float,             # fibre volume fraction (typ. 0.55–0.65 for aerospace prepreg)
) -> dict:
    """
    Halpin-Tsai micromechanics for unidirectional ply.
    Returns E1, E2, nu12, G12, rho for CLT input.

    Reference: Halpin & Kardos (1976); Jones (1998) "Mechanics of Composite Materials"
    """
    Vm = 1.0 - Vf

    # Rule of mixtures (longitudinal — fibre-dominated)
    E1  = fibre.E_f_GPa * Vf + matrix.E_m_GPa * Vm

    # Halpin-Tsai (transverse — matrix-dominated)
    xi  = 2.0    # reinforcement geometry factor (2 for circular fibres, E2)
    eta = (fibre.E_ft_GPa/matrix.E_m_GPa - 1) / (fibre.E_ft_GPa/matrix.E_m_GPa + xi)
    E2  = matrix.E_m_GPa * (1 + xi * eta * Vf) / (1 - eta * Vf)

    # Poisson (rule of mixtures)
    nu12 = fibre.nu_f * Vf + matrix.nu_m * Vm

    # Shear modulus (Halpin-Tsai, xi=1 for circular fibre shear)
    xi_s  = 1.0
    eta_s = (fibre.G_f_GPa/matrix.G_m_GPa - 1) / (fibre.G_f_GPa/matrix.G_m_GPa + xi_s)
    G12   = matrix.G_m_GPa * (1 + xi_s * eta_s * Vf) / (1 - eta_s * Vf)

    # Density (rule of mixtures, exact)
    rho = fibre.rho_kg_m3 * Vf + matrix.rho_kg_m3 * Vm

    return {
        "E1_GPa": round(E1, 2),
        "E2_GPa": round(E2, 2),
        "nu12":   round(nu12, 4),
        "G12_GPa": round(G12, 2),
        "rho_kg_m3": round(rho, 1),
        "Vf": Vf,
    }
```

## Step 2 — Classical Laminate Theory (ABD Matrix)

```python
def reduced_stiffness_matrix(E1, E2, nu12, G12):
    """Q matrix in principal material axes (GPa units → convert to Pa for SI)."""
    nu21 = nu12 * E2 / E1
    denom = 1 - nu12 * nu21
    Q = np.zeros((3, 3))
    Q[0,0] = E1 / denom
    Q[1,1] = E2 / denom
    Q[0,1] = Q[1,0] = nu12 * E2 / denom
    Q[2,2] = G12
    return Q  # GPa


def transform_Q(Q, theta_deg):
    """Rotate Q matrix to global x-y axes for ply at angle theta."""
    theta = np.radians(theta_deg)
    c, s  = np.cos(theta), np.sin(theta)
    T = np.array([
        [c**2,    s**2,    2*c*s],
        [s**2,    c**2,   -2*c*s],
        [-c*s,    c*s,  c**2-s**2],
    ])
    # Q_bar = T^-1 Q T^-T (Reuter matrix formulation)
    T_inv = np.linalg.inv(T)
    Reuter = np.diag([1, 1, 0.5])   # engineering → tensor shear
    return T_inv @ Q @ np.linalg.inv(T_inv).T


def build_ABD_matrix(
    ply_angles_deg: list[float],
    ply_thickness_mm: float,
    E1_GPa: float, E2_GPa: float, nu12: float, G12_GPa: float,
) -> dict:
    """
    Classical Laminate Theory ABD matrix.

    Returns 6×6 [A|B; B|D] matrix in [N/m, N, N·m] (SI, plies in mm).

    ply_angles_deg: list of ply angles, bottom to top (e.g., [0, 45, -45, 90, 90, -45, 45, 0])
    ply_thickness_mm: uniform ply thickness (typ. 0.125–0.25 mm for prepreg)
    """
    n_plies = len(ply_angles_deg)
    t_mm    = ply_thickness_mm
    H       = n_plies * t_mm   # total laminate thickness (mm)

    # Ply midplane z-coordinates (mm from midplane, bottom=-H/2)
    z = np.linspace(-H/2, H/2, n_plies + 1)

    Q   = reduced_stiffness_matrix(E1_GPa, E2_GPa, nu12, G12_GPa)  # GPa
    A   = np.zeros((3, 3))
    B   = np.zeros((3, 3))
    D   = np.zeros((3, 3))

    for k, theta in enumerate(ply_angles_deg):
        Qbar = transform_Q(Q, theta)  # GPa
        zk   = z[k+1]
        zk_1 = z[k]
        A   += Qbar * (zk - zk_1)                         # N/mm
        B   += Qbar * 0.5 * (zk**2 - zk_1**2)            # N
        D   += Qbar * (1/3) * (zk**3 - zk_1**3)          # N·mm

    # Convert to SI: A→N/m, B→N, D→N·m
    A *= 1e9 / 1e3    # GPa·mm → N/m (×10⁶, divide by 10³ for mm→m)
    B *= 1e9 / 1e6    # GPa·mm² → N (×10⁶, divide by 10⁶)
    D *= 1e9 / 1e9    # GPa·mm³ → N·m (×10⁶, divide by 10⁹)

    # Laminate engineering constants
    h = H * 1e-3   # m
    Ex  = (A[0,0]*A[1,1] - A[0,1]**2) / (A[1,1] * h)
    Ey  = (A[0,0]*A[1,1] - A[0,1]**2) / (A[0,0] * h)
    Gxy = A[2,2] / h
    nu_xy = A[0,1] / A[1,1]

    symmetric = np.allclose(B, 0, atol=1e-6)  # B≈0 for symmetric layup

    return {
        "A_N_m": A,      "B_N": B,       "D_N_m": D,
        "Ex_GPa": Ex/1e9, "Ey_GPa": Ey/1e9, "Gxy_GPa": Gxy/1e9,
        "nu_xy": nu_xy,
        "thickness_mm": H,
        "n_plies": n_plies,
        "is_symmetric": symmetric,
        "is_balanced": _check_balanced(ply_angles_deg),
    }


def _check_balanced(angles):
    """Balanced: equal +θ and -θ plies for A16=A26=0."""
    from collections import Counter
    c = Counter(angles)
    for angle in c:
        if angle not in [0, 90] and c[angle] != c.get(-angle, 0):
            return False
    return True
```

## Step 3 — Failure Analysis (Tsai-Wu)

```python
@dataclass
class PlyStrengths:
    """
    Ply strengths in principal directions. MUST cite source.
    Source: MMPDS, datasheet, or test report. Never recalled.
    """
    F1t_MPa: float    # longitudinal tensile
    F1c_MPa: float    # longitudinal compressive (positive value)
    F2t_MPa: float    # transverse tensile
    F2c_MPa: float    # transverse compressive (positive value)
    F12_MPa: float    # in-plane shear
    source: str       # mandatory provenance


def tsai_wu_failure_index(
    sigma1_MPa: float, sigma2_MPa: float, tau12_MPa: float,
    strengths: PlyStrengths,
) -> dict:
    """
    Tsai-Wu quadratic failure criterion.
    FI < 1.0 → no failure; FI = 1.0 → failure onset.

    Reference: Tsai & Wu (1971) J. Compos. Mater.

    Returns: FI, reserve_factor (RF = 1/√FI when FI>0), failure_mode
    """
    F1  = 1/strengths.F1t_MPa - 1/strengths.F1c_MPa
    F2  = 1/strengths.F2t_MPa - 1/strengths.F2c_MPa
    F11 = 1/(strengths.F1t_MPa * strengths.F1c_MPa)
    F22 = 1/(strengths.F2t_MPa * strengths.F2c_MPa)
    F66 = 1/strengths.F12_MPa**2
    F12 = -0.5 * np.sqrt(F11 * F22)  # interaction term (biaxial test needed; -0.5*√(F11F22) common)

    FI = (F1*sigma1_MPa + F2*sigma2_MPa +
          F11*sigma1_MPa**2 + F22*sigma2_MPa**2 +
          2*F12*sigma1_MPa*sigma2_MPa + F66*tau12_MPa**2)

    # Determine dominant failure mode
    terms = {
        "fiber_tension":  F11*sigma1_MPa**2 if sigma1_MPa > 0 else 0,
        "fiber_compression": F11*sigma1_MPa**2 if sigma1_MPa < 0 else 0,
        "matrix_tension": F22*sigma2_MPa**2 if sigma2_MPa > 0 else 0,
        "matrix_compression": F22*sigma2_MPa**2 if sigma2_MPa < 0 else 0,
        "shear": F66*tau12_MPa**2,
    }
    dominant = max(terms, key=terms.get)

    return {
        "FI": round(FI, 4),
        "reserve_factor": round(1/np.sqrt(max(FI, 1e-9)), 2),
        "fail": FI >= 1.0,
        "dominant_mode": dominant,
    }
```

## Step 4 — Interlaminar Shear and Delamination

```python
def interlaminar_shear_strength_check(
    Q_kN: float,           # transverse shear force per unit width (kN/m)
    h_mm: float,           # laminate thickness (mm)
    ILSS_MPa: float,       # interlaminar shear strength (from lap-shear test, ASTM D2344)
) -> dict:
    """
    Classical beam theory estimate of interlaminar shear stress.
    τ_max = 1.5 × V / (b·h) for rectangular cross-section.
    Valid for bending-dominated; check with FEA for complex joints.

    ILSS for carbon/epoxy: ~40–80 MPa (test required; do not recall).
    """
    tau_max_MPa = 1.5 * Q_kN * 1e3 / (1.0 * h_mm * 1e-3) / 1e6
    RF = ILSS_MPa / max(tau_max_MPa, 1e-6)
    return {
        "tau_ilss_MPa": round(tau_max_MPa, 3),
        "ILSS_MPa": ILSS_MPa,
        "reserve_factor": round(RF, 2),
        "status": "pass" if RF > 1.5 else "fail",
        "note": "Verify with ASTM D2344 or D7078 test data for final certification",
    }
```

## Manufacturing Constraints

For Swan topology-optimization output (anisotropic optimization):
- Ply angles from Swan result → round to manufacturable angles {0°, ±45°, 90°}
- Ply drops: maximum 1 ply per 6 mm step-back (avoid stress concentration)
- Minimum radius of curvature for prepreg layup: ≥ 3× ply thickness
- No resin-rich pockets: ensure fibres make contact on all inner radii
- Out-time limit: prepreg must be placed within manufacturer's specified out-time
- Autoclave cure cycle: define temperature/pressure ramp from material datasheet
