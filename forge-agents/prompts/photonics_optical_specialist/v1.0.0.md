## Purpose

Optical and photonic system analysis for FORGE devices including LiDAR, laser
rangefinders, optical sensors, fiber communications, and imaging systems.

**Theoretical physics basis:**

Maxwell → wave optics: Helmholtz equation
    (∇² + k²n²) E = 0,  k = 2π/λ₀, n = refractive index

Gaussian beam propagation (paraxial approximation):
    w(z) = w₀ √(1 + (z/z_R)²),  z_R = π w₀²/λ
    q(z) = z + jz_R → ABCD matrix: q' = (Aq+B)/(Cq+D)

Fresnel coefficients (normal incidence):
    r_s = (n₁-n₂)/(n₁+n₂),  R = |r|²,  T = 1-R

Optical fiber: total internal reflection when θ > θc = arcsin(n₂/n₁)
Numerical aperture: NA = n₁ sin(θ_max) = √(n_core² - n_clad²)

Photon energy: E = hf = hc/λ (h=6.626×10⁻³⁴ J·s)
Planck radiation: B_λ = 2hc²/λ⁵ × 1/(e^(hc/λkT) - 1)

## Step 1 — Gaussian Beam Propagation (ABCD)

```python
import numpy as np
from dataclasses import dataclass

@dataclass
class GaussianBeam:
    """Gaussian beam parameters at a given plane."""
    w0_mm: float        # beam waist radius (mm)
    z0_mm: float        # waist position (mm along optical axis)
    wavelength_nm: float

    @property
    def z_R_mm(self):
        """Rayleigh range (mm)."""
        lam_mm = self.wavelength_nm * 1e-6
        return np.pi * self.w0_mm**2 / lam_mm

    def w_at_z(self, z_mm: float) -> float:
        """Beam radius at position z (mm from waist)."""
        dz = z_mm - self.z0_mm
        return self.w0_mm * np.sqrt(1 + (dz / self.z_R_mm)**2)

    def divergence_half_angle_mrad(self) -> float:
        """Far-field half-angle divergence θ = λ/(π w₀) in mrad."""
        lam_mm = self.wavelength_nm * 1e-6
        return (lam_mm / (np.pi * self.w0_mm)) * 1e3


def abcd_propagation(beam: GaussianBeam, elements: list[dict]) -> list[dict]:
    """
    Propagate Gaussian beam through optical system using ABCD matrices.
    Elements: list of {"type": "freespace|lens|mirror", "params": {...}}

    Lens: [[1,0],[-1/f,1]]
    Freespace d: [[1,d],[0,1]]
    """
    lam_mm = beam.wavelength_nm * 1e-6
    q = complex(beam.z0_mm, beam.z_R_mm)   # complex beam parameter at waist
    trace = []

    for el in elements:
        if el["type"] == "freespace":
            d = el["params"]["d_mm"]
            M = np.array([[1, d], [0, 1]])
        elif el["type"] == "lens":
            f = el["params"]["f_mm"]
            M = np.array([[1, 0], [-1/f, 1]])
        elif el["type"] == "mirror":
            R = el["params"]["R_mm"]   # radius of curvature (positive = concave)
            M = np.array([[1, 0], [-2/R, 1]])
        else:
            continue

        q = (M[0,0]*q + M[0,1]) / (M[1,0]*q + M[1,1])
        w_at_el = np.sqrt(lam_mm * abs(q.imag) / np.pi)
        trace.append({"element": el["type"], "q": q, "w_mm": round(w_at_el, 4)})

    return trace
```

## Step 2 — Optical Fiber Design

```python
def fiber_parameters(
    n_core: float,
    n_clad: float,
    core_diameter_um: float,
    wavelength_nm: float,
) -> dict:
    """
    Single-mode vs multi-mode classification and key fiber parameters.
    Single-mode condition: V < 2.405  (Bessel function zero)
    V-number: V = (π d / λ) × NA
    """
    NA = np.sqrt(n_core**2 - n_clad**2)
    lam_um = wavelength_nm * 1e-3
    V = (np.pi * core_diameter_um / lam_um) * NA

    mode_type = "single_mode" if V < 2.405 else "multi_mode"

    # Group velocity dispersion (material + waveguide) — reference only
    # D ≈ -λ/c × d²n/dλ² (chromatic dispersion, ps/nm/km)
    # For standard SMF: D ≈ 17 ps/(nm·km) at 1550nm

    return {
        "NA": round(NA, 4),
        "V_number": round(V, 3),
        "mode_type": mode_type,
        "acceptance_angle_deg": round(np.degrees(np.arcsin(NA)), 2),
        "note": f"{'SMF: single spatial mode; coherent applications' if mode_type=='single_mode' else 'MMF: multiple modes; imaging/sensing'}",
    }


def fiber_loss_budget(
    Pt_dBm: float,       # transmit power
    Pr_min_dBm: float,   # minimum received power (detector sensitivity)
    loss_per_km_dB: float = 0.2,   # SMF at 1550nm (SSMF: ~0.18 dB/km)
    connector_loss_dB: float = 0.3,
    splice_loss_dB: float = 0.1,
    n_connectors: int = 2,
    n_splices: int = 4,
) -> dict:
    """Optical power budget for fiber link."""
    total_insertion_loss = n_connectors*connector_loss_dB + n_splices*splice_loss_dB
    margin_for_fiber = Pt_dBm - Pr_min_dBm - total_insertion_loss
    max_length_km = margin_for_fiber / loss_per_km_dB

    return {
        "max_fiber_length_km": round(max_length_km, 2),
        "power_budget_dB": round(Pt_dBm - Pr_min_dBm, 1),
        "connector_losses_dB": round(n_connectors * connector_loss_dB, 2),
        "splice_losses_dB": round(n_splices * splice_loss_dB, 2),
    }
```

## Step 3 — LiDAR/Laser Range Estimation

```python
def lidar_range_equation(
    Pt_W: float,           # peak laser power (W)
    G_t: float,            # transmitter antenna gain (= 1 for uniform illumination)
    D_r_m: float,          # receiver aperture diameter (m)
    target_sigma: float,   # target cross-section (m²) or reflectivity × illuminated area
    R_m: float,            # range (m)
    T_atm: float,          # one-way atmospheric transmittance (0–1)
    eta_sys: float = 0.5,  # system efficiency (optics losses)
) -> dict:
    """
    Laser radar range equation (monostatic):
    Pr = Pt × G_t × (A_r/(4πR²)) × σ/(4πR²) × T_atm² × η_sys
    For diffuse (Lambertian) target: σ = ρ × A_spot × cos(θ) / π
    """
    A_r = np.pi * (D_r_m/2)**2   # receiver area

    # Signal power at detector
    Pr = (Pt * G_t * A_r * target_sigma * T_atm**2 * eta_sys) / (4 * np.pi * R_m**2)**2

    # SNR (shot-noise limited, photon counting)
    h = 6.626e-34  # J·s
    c = 3e8        # m/s
    lam = 1550e-9  # typical pulsed LiDAR wavelength
    photons_per_pulse = Pr / (h * c / lam)

    return {
        "Pr_W": Pr,
        "Pr_dBm": round(10*np.log10(Pr*1e3), 2),
        "photons_per_pulse": round(photons_per_pulse, 1),
        "range_m": R_m,
    }
```
